    for (final option in avatars) {

/// Вороненок: парит и моргает (animated: true) или стоит неподвижно.
class ScivoRaven extends StatefulWidget {
  const ScivoRaven({
    super.key,
    this.size = 160,
    this.avatarId = 'raven_blue',
    this.animated = true,
  });

  final double size;
  final String avatarId;
  final bool animated;

  @override
  State<ScivoRaven> createState() => _ScivoRavenState();
}

class _ScivoRavenState extends State<ScivoRaven>
    with SingleTickerProviderStateMixin {
  AnimationController? _bob;
  Timer? _blinkTimer;
  bool _eyesClosed = false;

  @override
  void initState() {
    super.initState();
    if (widget.animated) {
      _bob = AnimationController(
        vsync: this,
        duration: const Duration(milliseconds: 1600),
      )..repeat(reverse: true);
      _scheduleBlink();
    }
  }

  void _scheduleBlink() {
    _blinkTimer = Timer(const Duration(milliseconds: 3200), () {
      if (!mounted) return;
      setState(() => _eyesClosed = true);
      _blinkTimer = Timer(const Duration(milliseconds: 160), () {
        if (!mounted) return;
        setState(() => _eyesClosed = false);
        _scheduleBlink();
      });
    });
  }

  @override
  void dispose() {
    _blinkTimer?.cancel();
    _bob?.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    final picture = SvgPicture.string(
      RavenArt.svg(widget.avatarId, blink: _eyesClosed),
      width: widget.size,
      height: widget.size,
    );
    final bob = _bob;
    if (bob == null) return picture;

    return AnimatedBuilder(
      animation: bob,
      builder: (context, child) {
        final t = Curves.easeInOut.transform(bob.value);
        return Transform.translate(offset: Offset(0, -8 * t), child: child);
      },
      child: picture,
    );
  }
}

class _StarsPainter extends CustomPainter {
  const _StarsPainter();

  @override
  void paint(Canvas canvas, Size size) {
    final random = math.Random(7);
    final paint = Paint();
    for (var i = 0; i < 45; i++) {
      final dx = random.nextDouble() * size.width;
      final dy = random.nextDouble() * size.height;
      final radius = 0.6 + random.nextDouble() * 1.6;
      paint.color = Colors.white.withValues(alpha: 0.15 + random.nextDouble() * 0.45);
      canvas.drawCircle(Offset(dx, dy), radius, paint);
    }
  }

  @override
  bool shouldRepaint(covariant CustomPainter oldDelegate) => false;
}

/// Русское склонение: pluralRu(5, 'перо', 'пера', 'перьев') -> 'перьев'.
String pluralRu(int n, String one, String few, String many) {
  final mod10 = n % 10;
  final mod100 = n % 100;
  if (mod10 == 1 && mod100 != 11) return one;
  if (mod10 >= 2 && mod10 <= 4 && (mod100 < 12 || mod100 > 14)) return few;
  return many;
}

import 'package:flutter/material.dart';

import '../core/theme/app_theme.dart';

/// Крупная цифровая клавиатура для PIN.
/// onCompleted возвращает true (PIN принят) или false (очистить и ввести заново).
class PinPad extends StatefulWidget {
  const PinPad({super.key, required this.onCompleted, this.length = 4});

  final Future<bool> Function(String pin) onCompleted;
  final int length;

  @override
  State<PinPad> createState() => _PinPadState();
}

class _PinPadState extends State<PinPad> {
  String _pin = '';
  bool _busy = false;

  Future<void> _tapDigit(String digit) async {
    if (_busy || _pin.length >= widget.length) return;
    setState(() => _pin = _pin + digit);
    if (_pin.length < widget.length) return;

    setState(() => _busy = true);
    var ok = false;
    try {
      ok = await widget.onCompleted(_pin);
    } catch (_) {
      ok = false;
    }
    if (!mounted) return;
    setState(() {
      _busy = false;
      if (!ok) _pin = '';
    });
  }

  void _backspace() {
    if (_busy || _pin.isEmpty) return;
    setState(() => _pin = _pin.substring(0, _pin.length - 1));
  }

  @override
  Widget build(BuildContext context) {
    const rows = [
      ['1', '2', '3'],
      ['4', '5', '6'],
      ['7', '8', '9'],
      ['', '0', '<'],
    ];

    return Column(
      mainAxisSize: MainAxisSize.min,
      children: [
        Row(
          mainAxisSize: MainAxisSize.min,
          children: [
            for (var i = 0; i < widget.length; i++)
              AnimatedContainer(
                duration: const Duration(milliseconds: 150),
                margin: const EdgeInsets.symmetric(horizontal: 10),
                width: 22,
                height: 22,
                decoration: BoxDecoration(
                  shape: BoxShape.circle,
                  color: i < _pin.length ? AppColors.gold : Colors.transparent,
                  border: Border.all(color: AppColors.gold, width: 2.5),
                ),
              ),
          ],
        ),
        const SizedBox(height: 12),
        SizedBox(
          height: 4,
          width: 120,
          child: _busy ? const LinearProgressIndicator() : null,
        ),
        const SizedBox(height: 8),
        for (final row in rows)
          Row(
            mainAxisSize: MainAxisSize.min,
            children: [
              for (final key in row)
                Padding(
                  padding: const EdgeInsets.all(7),
                  child: key.isEmpty
                      ? const SizedBox(width: 74, height: 74)
                      : _PinKey(
                          onTap: key == '<' ? _backspace : () => _tapDigit(key),
                          child: key == '<'
                              ? const Icon(Icons.backspace_rounded, size: 28, color: Colors.white)
                              : Text(
                                  key,
                                  style: const TextStyle(
                                    fontSize: 30,
                                    fontWeight: FontWeight.w800,
                                    color: Colors.white,
                                  ),
                                ),
                        ),
                ),
            ],
          ),
      ],
    );
  }
}

class _PinKey extends StatelessWidget {
  const _PinKey({required this.onTap, required this.child});

  final VoidCallback onTap;

      ),
    );
  }
}
