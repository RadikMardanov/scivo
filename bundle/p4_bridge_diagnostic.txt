/// Задача, выданная сервером (generate_task).
class GeneratedTask {
  const GeneratedTask({
    required this.taskId,
    required this.worldId,
    required this.mechanic,
    required this.difficulty,
    required this.payload,
  });

  final String taskId;
  final String worldId;
  final String mechanic;
  final int difficulty;
  final Map<String, dynamic> payload;

  factory GeneratedTask.fromJson(Map<String, dynamic> json) {
    // DIAG_BRIDGE [PARSER]: логируем сырой ответ сервера перед парсингом.
    try {
      // ignore: avoid_print
      print('DIAG_BRIDGE [PARSER] raw_json_keys=${json.keys.toList()}');
      // ignore: avoid_print
      print('DIAG_BRIDGE [PARSER] raw_payload_type=${json['payload'].runtimeType}');
      // ignore: avoid_print
      print('DIAG_BRIDGE [PARSER] raw_payload_value=${json['payload']}');
    } catch (_) {/* safety */}

    // ИСПРАВЛЕНО (R-05): надёжное извлечение payload.
    // Supabase может вернуть payload как Map<dynamic,dynamic>, LinkedHashMap,
    // _JsonMap или даже null при ошибке десериализации. Приводим ко всему,
    // что похоже на словарь, иначе — пустой map.
    Map<String, dynamic> safePayload;
    final raw = json['payload'];
    if (raw is Map) {
      safePayload = Map<String, dynamic>.from(raw);
    } else if (raw is String && raw.isNotEmpty) {
      // На случай двойного JSON-кодирования (редко, но бывает).
      try {
        final decoded = Uri.decodeComponent(raw);
        safePayload = <String, dynamic>{'__raw_string__': decoded};
      } catch (_) {
        safePayload = <String, dynamic>{'__raw_string__': raw};
      }
    } else {
      safePayload = <String, dynamic>{};
    }

    // DIAG_BRIDGE [PARSER]: логируем итоговый payload после приведения.
    try {
      // ignore: avoid_print
      print('DIAG_BRIDGE [PARSER] parsed_payload_keys=${safePayload.keys.toList()}');
      // ignore: avoid_print
      print('DIAG_BRIDGE [PARSER] parsed_payload=$safePayload');
    } catch (_) {/* safety */}

    return GeneratedTask(
      taskId: json['task_id']?.toString() ?? '',
      worldId: json['world_id']?.toString() ?? '',
      mechanic: json['mechanic']?.toString() ?? Mechanics.keypad,
      difficulty: (json['difficulty'] as num?)?.toInt() ?? 1,
      payload: safePayload,
    );
  }
}

/// Награда за попытку (submit_answer -> reward).
class Reward {
  const Reward({
    required this.feather,
    required this.mana,
    required this.xp,
    required this.fixBonus,
  });

  final int feather;
  final int mana;
  final int xp;
  final int fixBonus;

  static const zero = Reward(feather: 0, mana: 0, xp: 0, fixBonus: 0);

  factory Reward.fromJson(Map<String, dynamic> json) {
    return Reward(
      feather: (json['feather'] as num?)?.toInt() ?? 0,
      mana: (json['mana'] as num?)?.toInt() ?? 0,
      xp: (json['xp'] as num?)?.toInt() ?? 0,
      fixBonus: (json['fix_bonus'] as num?)?.toInt() ?? 0,
    );
  }
}

/// Результат проверки ответа сервером (submit_answer).
class SubmitResult {
  const SubmitResult({
    required this.correct,
    required this.reward,
    required this.feedbackKey,
    required this.hintLevel,
  });

  final bool correct;
  final Reward reward;
  final String feedbackKey; // success_clean | success_hinted | wrong
  final int hintLevel;

  factory SubmitResult.fromJson(Map<String, dynamic> json) {
    return SubmitResult(
      correct: json['correct'] == true,
      reward: Reward.fromJson(
        Map<String, dynamic>.from(json['reward'] as Map? ?? const {}),
      ),
      feedbackKey: json['feedback_key']?.toString() ?? 'wrong',
      hintLevel: (json['hint_level'] as num?)?.toInt() ?? 0,
    );
  }
}

/// Результат запроса подсказки (request_hint).
class HintResult {
  const HintResult({
    required this.text,
    required this.level,
    required this.manaSpent,
  });

  final String text;
  final int level;
  final int manaSpent;

  factory HintResult.fromJson(Map<String, dynamic> json) {
    final hint = json['hint'];
    final text = hint is Map
        ? (hint['text']?.toString() ?? '')
        : (hint?.toString() ?? '');
    return HintResult(
      text: text.isEmpty
          ? 'Подсказка готовится. Попробуй ещё раз, я рядом!'
          : text,
      level: (json['level'] as num?)?.toInt() ?? 0,
      manaSpent: (json['mana_spent'] as num?)?.toInt() ?? 0,
    );
  }
}

import 'dart:math' as math;

/// Рисует УСЛОВИЕ задачи (то, что видит ребёнок). Правильного ответа тут нет —
/// он живёт только на сервере в active_tasks.answer.
class TaskVisual extends StatelessWidget {
  const TaskVisual({super.key, required this.payload});
  final Map<String, dynamic> payload;

  int _i(dynamic v, [int d = 0]) => (v as num?)?.toInt() ?? d;
  String _s(dynamic v) => v?.toString() ?? '';

  @override
  Widget build(BuildContext context) {
    // DIAG_BRIDGE [VISUAL]: логируем то, что реально получил визуализатор.
    try {
      // ignore: avoid_print
      print('DIAG_BRIDGE [VISUAL] payload_keys=${payload.keys.toList()}');
      // ignore: avoid_print
      print('DIAG_BRIDGE [VISUAL] kind=${payload['kind']} a=${payload['a']} b=${payload['b']} op=${payload['op']}');
    } catch (_) {/* safety */}

    // Пустой payload — ранний возврат с понятной заглушкой, без «0+0=?».
    if (payload.isEmpty) {
      return const Padding(
        padding: EdgeInsets.symmetric(vertical: 24),
        child: Center(
          child: Text(
            'Задача загружается… 🐦⬛',
            style: TextStyle(fontSize: 20, color: Colors.white70),
          ),
        ),
      );
    }

    final kind = _s(payload['kind']);

    // Основной путь: dispatch по полю kind.
    if (kind.isNotEmpty) {
      switch (kind) {
        case 'mul':
          return _Equation('${_i(payload['a'])} × ${_i(payload['b'])}');
        case 'arith':
          final opSym = _s(payload['op']).toLowerCase() == 'sub' ? '−' : '+';
          return _Equation('${_i(payload['a'])} $opSym ${_i(payload['b'])}');
        case 'add':
          return _Equation('${_i(payload['a'])} + ${_i(payload['b'])}');
        case 'sub':
          return _Equation('${_i(payload['a'])} − ${_i(payload['b'])}');
        case 'read_clock':
          return _ClockFace(hour: _i(payload['hour']), minute: _i(payload['minute']));
        case 'compare_frac':
          return _CompareFractions(
            leftNum: _i(payload['left_num']),
            rightNum: _i(payload['right_num']),
            den: _i(payload['den'], 4),
          );
        case 'compose':
          return _ComposeHint(total: _i(payload['total'], 10), given: _i(payload['given']));
        case 'shade_frac':
          return _ShadeHint(parts: _i(payload['parts'], 8), need: _i(payload['need'], 1));
        default:
          break; // падаем в структурный fallback ниже
      }
    }

    // Структурный fallback: определяем тип по наличию полей, если kind потерян.
    if (payload.containsKey('a') && payload.containsKey('b')) {
      if (payload.containsKey('options')) {
        return _Equation('${_i(payload['a'])} × ${_i(payload['b'])}');
      }
      final opSym = _s(payload['op']).toLowerCase() == 'sub' ? '−' : '+';
      return _Equation('${_i(payload['a'])} $opSym ${_i(payload['b'])}');
    }
    if (payload.containsKey('hour') && payload.containsKey('minute')) {
      return _ClockFace(hour: _i(payload['hour']), minute: _i(payload['minute']));
    }
    if (payload.containsKey('left_num') && payload.containsKey('den')) {
      return _CompareFractions(
        leftNum: _i(payload['left_num']),
        rightNum: _i(payload['right_num']),
        den: _i(payload['den'], 4),
      );
    }
    if (payload.containsKey('total') && payload.containsKey('given')) {
      return _ComposeHint(total: _i(payload['total'], 10), given: _i(payload['given']));
    }
    if (payload.containsKey('parts') && payload.containsKey('need')) {
      return _ShadeHint(parts: _i(payload['parts'], 8), need: _i(payload['need'], 1));
    }

    // Если ничего не распознали — диагностический вывод (не бросаем исключение).
    return Padding(
      padding: const EdgeInsets.all(12),
      child: Text(
        'DEBUG: keys=${payload.keys.join(', ')}\nvalues=${payload.values.map((v) => v.toString()).join(', ')}',
        textAlign: TextAlign.center,
        style: const TextStyle(fontSize: 12, color: Colors.orangeAccent),
      ),
    );
  }
}

class _Equation extends StatelessWidget {
  const _Equation(this.text);
  final String text;

  @override
  Widget build(BuildContext context) {
    // Защита от вырожденных значений: если текст выглядит как "0 + 0" или "0 × 0",
    // это почти наверняка признак потери данных — предупреждаем в лог.
    final looksEmpty = RegExp(r'^\s*0\s*[+\-−×]\s*0\s*$').hasMatch(text);
    if (looksEmpty) {
      // ignore: avoid_print
      print('DIAG_BRIDGE [VISUAL_WARN] suspicious_zero_equation="$text"');
    }
    return Padding(
      padding: const EdgeInsets.symmetric(vertical: 16),
      child: Text(
        '$text = ?',
        textAlign: TextAlign.center,
        style: const TextStyle(
          fontSize: 40,
          fontWeight: FontWeight.w900,
          color: AppColors.gold,
        ),
      ),
    );
  }
}

class _ClockPainter extends CustomPainter {
  _ClockPainter({required this.hour, required this.minute});
  final int hour, minute;

  @override
  void paint(Canvas canvas, Size size) {
    final c = Offset(size.width / 2, size.height / 2);
    final r = size.width / 2 - 6;
    canvas.drawCircle(
        c,
        r,
        Paint()
          ..style = PaintingStyle.stroke
          ..strokeWidth = 4
          ..color = AppColors.purpleLight);
    final tick = Paint()
      ..color = Colors.white70
      ..strokeWidth = 3;
    for (var i = 0; i < 12; i++) {
      final a = (math.pi / 6) * i;
      canvas.drawLine(
        c + Offset(math.sin(a), -math.cos(a)) * (r - 10),
        c + Offset(math.sin(a), -math.cos(a)) * r,
        tick,
      );
    }
    final hAng = (math.pi / 6) * ((hour % 12) + minute / 60);
    final mAng = (math.pi / 30) * minute;
    canvas.drawLine(
      c,
      c + Offset(math.sin(hAng), -math.cos(hAng)) * (r * 0.5),
      Paint()
        ..color = Colors.white
        ..strokeWidth = 6
        ..strokeCap = StrokeCap.round,
    );
    canvas.drawLine(
      c,
      c + Offset(math.sin(mAng), -math.cos(mAng)) * (r * 0.78),
      Paint()
        ..color = AppColors.gold
        ..strokeWidth = 4
        ..strokeCap = StrokeCap.round,
    );
    canvas.drawCircle(c, 6, Paint()..color = AppColors.gold);
  }

  @override
  bool shouldRepaint(covariant _ClockPainter old) =>
      old.hour != hour || old.minute != minute;
}

import 'package:flutter_riverpod/flutter_riverpod.dart';

/// Состояние экрана квеста: текущая задача, результат последней попытки,
/// уровень использованных подсказок, статус загрузки, текст ошибки.
class QuestState {
  const QuestState({
    this.task,
    this.lastResult,
    this.lastHint,
    this.hintLevel = 0,
    this.isLoading = false,
    this.error,
  });

  final GeneratedTask? task;
  final SubmitResult? lastResult;
  final HintResult? lastHint;
  final int hintLevel;
  final bool isLoading;
  final String? error;

  QuestState copyWith({
    GeneratedTask? task,
    SubmitResult? lastResult,
    HintResult? lastHint,
    int? hintLevel,
    bool? isLoading,
    String? error,
  }) {
    return QuestState(
      task: task ?? this.task,
      lastResult: lastResult ?? this.lastResult,
      lastHint: lastHint ?? this.lastHint,
      hintLevel: hintLevel ?? this.hintLevel,
      isLoading: isLoading ?? this.isLoading,
      error: error,
    );
  }
}

/// Контроллер квеста: вызывает RPC generate_task / submit_answer / request_hint.
class QuestController extends StateNotifier<QuestState> {
  QuestController(this._service) : super(const QuestState());

  final SupabaseService _service;

  Future<void> loadTask(String childId, String worldId) async {
    state = state.copyWith(
      isLoading: true,
      error: null,
      lastResult: null,
      lastHint: null,
      hintLevel: 0,
    );
    try {
      final task = await _service.generateTask(childId, worldId);
      // DIAG_BRIDGE [CONTROLLER]: фиксируем момент получения задачи контроллером.
      // ignore: avoid_print
      print('DIAG_BRIDGE [CONTROLLER] loaded task_id=${task.taskId} world=${task.worldId} mech=${task.mechanic} diff=${task.difficulty}');
      // ignore: avoid_print
      print('DIAG_BRIDGE [CONTROLLER] payload_in_controller=${task.payload}');
      state = state.copyWith(task: task, isLoading: false);
    } catch (e, st) {
      // ignore: avoid_print
      print('DIAG_BRIDGE [CONTROLLER_ERROR] $e\n$st');
      state = state.copyWith(error: e.toString(), isLoading: false);
    }
  }

  Future<void> submit(String childId, Map<String, dynamic> input) async {
    final task = state.task;
    if (task == null) return;

    state = state.copyWith(isLoading: true);
    try {
      final result = await _service.submitAnswer(childId, task.taskId, input);
      state = state.copyWith(lastResult: result, isLoading: false);
    } catch (e) {
      state = state.copyWith(error: e.toString(), isLoading: false);
    }
  }

  Future<void> askHint(String childId, int level) async {
    final task = state.task;
    if (task == null) return;

    state = state.copyWith(isLoading: true);
    try {
      final hint = await _service.requestHint(childId, task.taskId, level);
      state = state.copyWith(lastHint: hint, hintLevel: level, isLoading: false);
    } catch (e) {
      state = state.copyWith(error: e.toString(), isLoading: false);
    }
  }

  void reset() {
    state = const QuestState();
  }
}
