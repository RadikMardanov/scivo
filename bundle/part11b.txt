/// Экран редактирования имени и класса ребёнка.
class EditChildScreen extends ConsumerStatefulWidget {
  const EditChildScreen({super.key, required this.child});
  final ChildProfile child;

  @override
  ConsumerState<EditChildScreen> createState() => _EditChildScreenState();
}

class _EditChildScreenState extends ConsumerState<EditChildScreen> {
  late TextEditingController _nameController;
  late int _selectedGrade;
  bool _isLoading = false;

  @override
  void initState() {
    super.initState();
    _nameController = TextEditingController(text: widget.child.nickname);
    _selectedGrade = widget.child.grade;
  }

  @override
  void dispose() {
    _nameController.dispose();
    super.dispose();
  }

  Future<void> _save() async {
    setState(() => _isLoading = true);
    try {
      final service = ref.read(serviceProvider);
      await service.updateChild(
        childId: widget.child.id,
        nickname: _nameController.text.trim(),
        grade: _selectedGrade,
      );
      // Обновляем список детей
      ref.invalidate(childrenProvider);
      if (mounted) Navigator.pop(context, true);
    } catch (e) {
      ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text(e.toString())));
    } finally {
      if (mounted) setState(() => _isLoading = false);
    }
  }

  @override
  Widget build(BuildContext context) {
    return MagicScaffold(
      title: 'Редактировать героя',
      body: SingleChildScrollView(
        padding: const EdgeInsets.all(20),
        child: Column(
          crossAxisAlignment: CrossAxisAlignment.stretch,
          children: [
            Text('Имя героя:', style: Theme.of(context).textTheme.titleMedium),
            const SizedBox(height: 8),
            TextField(
              controller: _nameController,
              maxLength: 20,
              decoration: InputDecoration(
                hintText: 'Например, Тагир',
                counterText: '',
                border: OutlineInputBorder(borderRadius: BorderRadius.circular(12)),
              ),
            ),
            const SizedBox(height: 24),
            Text('Класс:', style: Theme.of(context).textTheme.titleMedium),
            const SizedBox(height: 12),
            ...List.generate(4, (index) {
              final grade = index + 1;
              return RadioListTile<int>(
                value: grade,
                groupValue: _selectedGrade,
                onChanged: (val) => setState(() => _selectedGrade = val!),
                title: Text('$grade класс'),
                controlAffinity: ListTileControlAffinity.trailing,
              );
            }),
            const SizedBox(height: 32),
            FilledButton.icon(
              onPressed: _isLoading ? null : _save,
              icon: _isLoading
                  ? const SizedBox(width: 18, height: 18, child: CircularProgressIndicator(strokeWidth: 2))
                  : const Icon(Icons.save_rounded),
              label: Text(_isLoading ? 'Сохраняем...' : 'Сохранить изменения'),
              style: FilledButton.styleFrom(
                minimumSize: const Size(double.infinity, 56),
              ),
            ),
          ],
        ),
      ),
    );
  }
}

class SupabaseService {
  SupabaseService(this._client);

  final SupabaseClient _client;

  Session? get currentSession => _client.auth.currentSession;
  Stream<AuthState> get authChanges => _client.auth.onAuthStateChange;

  // ---------- Аккаунт родителя ----------
  Future<void> signUpParent({
    required String email,
    required String password,
    required String displayName,
  }) async {
    await _client.auth.signUp(
      email: email.trim(),
      password: password,
      data: {
        'display_name': displayName.trim(),
        'consent': 'true',
      },
    );
  }

  Future<void> signIn(String email, String password) async {
    await _client.auth.signInWithPassword(
      email: email.trim(),
      password: password,
    );
  }

  Future<void> signOut() => _client.auth.signOut();

  Future<ParentProfile?> fetchParent() async {
    final userId = _client.auth.currentUser?.id;
    if (userId == null) return null;
    final row = await _client
        .from('parents')
        .select('id, display_name, has_pin, consent_at')
        .eq('id', userId)
        .maybeSingle();
    return row == null ? null : ParentProfile.fromJson(row);
  }

  Future<void> giveConsent() async {
    await _client.rpc('give_consent');
  }

  Future<void> setParentPin(String pin) async {
    final result =
        await _client.rpc('set_parent_pin', params: {'p_new_pin': pin});
    final check = PinCheckResult.fromJson(_asMap(result));
    if (!check.ok) throw AppException(check.message);
  }

  Future<PinCheckResult> verifyParentPin(String pin) async {
    final result =
        await _client.rpc('verify_parent_pin', params: {'p_pin': pin});
    return PinCheckResult.fromJson(_asMap(result));
  }

  // ---------- Дети ----------
  Future<List<ChildProfile>> fetchChildren() async {
    final rows = await _client
        .from('children')
        .select(
            'id, nickname, avatar_id, grade, onboarding_done, daily_feather_cap')
        .order('created_at');
    return rows.map((row) => ChildProfile.fromJson(row)).toList();
  }

  Future<String> createChild({
    required String nickname,
    required String avatarId,
    required int grade,
    required String pin,
  }) async {
    final result = await _client.rpc('create_child', params: {
      'p_nickname': nickname.trim(),
      'p_avatar_id': avatarId,
      'p_grade': grade,
      'p_pin': pin,
    });
    return result.toString();
  }

  /// Обновляет имя и класс ребёнка (без PIN).
  Future<void> updateChild({
    required String childId,
    required String nickname,
    required int grade,
  }) async {
    await _client.rpc('update_child_profile', params: {
      'p_child_id': childId,
      'p_nickname': nickname.trim(),
      'p_grade': grade,
    });
  }

  Future<PinCheckResult> verifyChildPin(String childId, String pin) async {
    final result = await _client.rpc('verify_child_pin', params: {
      'p_child_id': childId,
      'p_pin': pin,
    });
    return PinCheckResult.fromJson(_asMap(result));
  }

  // ---------- Прогресс ----------
  Future<ChildProgress> fetchProgress(String childId) async {
    final row = await _client
        .from('child_progress')
        .select()
        .eq('child_id', childId)
        .maybeSingle();
    return row == null ? ChildProgress.empty : ChildProgress.fromJson(row);
  }

  Future<CheckinResult> checkin(String childId) async {
    final raw =
        await _client.rpc('child_checkin', params: {'p_child_id': childId});
    final result = _asMap(raw);
    return CheckinResult(
      progress:
          ChildProgress.fromJson(_asMap(result['progress'] as Map? ?? const {})),
      manaBonus: (result['mana_bonus'] as num?)?.toInt() ?? 0,
      usedFreeze: result['used_freeze'] == true,
    );

    final raw = await _client

    final result = _asMap(raw);

class DragInput extends StatefulWidget {
  const DragInput({super.key, required this.total, required this.given, required this.enabled, required this.onSubmit});

  final int total;
  final int given;
  final bool enabled;
  final void Function(Map<String, dynamic>) onSubmit;

  State<DragInput> createState() => _DragInputState();

class _DragInputState extends State<DragInput> {
  late int _rightCount;

  @override
  void initState() {
    super.initState();
    // ИСПРАВЛЕНО: начинаем с нуля, а не с правильного ответа
    _rightCount = 0;
  }

  void _tapRight(int delta) {
    if (!widget.enabled) return;
    setState(() {
      _rightCount = (_rightCount + delta).clamp(0, widget.total - widget.given);
    });
  }

  void _send() {
    if (!widget.enabled) return;
    widget.onSubmit({'right': _rightCount});
  }

  @override
  Widget build(BuildContext context) {
    return Column(
      children: [
        Row(
          mainAxisAlignment: MainAxisAlignment.spaceEvenly,
          children: [
            _Basket(label: 'Левая (${widget.given})', count: widget.given, color: AppColors.purpleLight),
            _Basket(label: 'Правая ($_rightCount)', count: _rightCount, color: AppColors.gold),
          ],
        ),
        const SizedBox(height: 16),
        Row(
          mainAxisAlignment: MainAxisAlignment.center,
          children: [
            IconButton(onPressed: () => _tapRight(-1), icon: const Icon(Icons.remove_circle_outline_rounded)),
            const SizedBox(width: 16),
            TextButton(onPressed: _send, child: const Text('Готово ✨')),
            const SizedBox(width: 16),
            IconButton(onPressed: () => _tapRight(1), icon: const Icon(Icons.add_circle_outline_rounded)),
          ],
        ),
      ],
    );
  }
}

/// Механика «закрашивание долей»: пицца/круг делится на части, ребёнок тапает нужные.
class ShadeInput extends StatefulWidget {
  const ShadeInput({super.key, required this.parts, required this.need, required this.enabled, required this.onSubmit});

  final int parts;
  final int need;
  final bool enabled;
  final void Function(Map<String, dynamic>) onSubmit;

  @override
  State<ShadeInput> createState() => _ShadeInputState();
}

class _ShadeInputState extends State<ShadeInput> {
  late Set<int> _shaded;

  @override
  void initState() {
    super.initState();
    _shaded = {};
  }

  void _toggle(int index) {
    if (!widget.enabled) return;
    setState(() {
      if (_shaded.contains(index)) {
        _shaded.remove(index);
      } else if (_shaded.length < widget.need) {
        _shaded.add(index);
      }
    });
  }

  void _send() {
    if (!widget.enabled) return;
    widget.onSubmit({'shaded': _shaded.toList()..sort()});
  }

  @override
  Widget build(BuildContext context) {
    final p = widget.parts.clamp(1, 12);
    return Column(
      children: [
        SizedBox(
          height: MediaQuery.sizeOf(context).width * 0.8, // адаптивная высота
          child: GridView.builder(
            physics: const NeverScrollableScrollPhysics(),
            gridDelegate: SliverGridDelegateWithFixedCrossAxisCount(
              crossAxisCount: p <= 4 ? 2 : 3,
              mainAxisSpacing: 8,
              crossAxisSpacing: 8,
            ),
            itemCount: p,
            itemBuilder: (_, i) => GestureDetector(
              onTap: () => _toggle(i),
              child: AnimatedContainer(
                duration: const Duration(milliseconds: 200),
                decoration: BoxDecoration(
                  color: _shaded.contains(i) ? AppColors.gold : Colors.white.withValues(alpha: 0.1),
                  borderRadius: BorderRadius.circular(12),
                  border: Border.all(color: _shaded.contains(i) ? AppColors.textDark : Colors.white24, width: 2),
                ),
                alignment: Alignment.center,
                child: Text('${i + 1}', style: TextStyle(
                  fontSize: 24,
                  fontWeight: FontWeight.w900,
                  color: _shaded.contains(i) ? AppColors.textDark : Colors.white70,
                )),
              ),
            ),
          ),
        ),
        const SizedBox(height: 12),
        Text('Закрась ${widget.need} из $p', textAlign: TextAlign.center),

import 'dart:math' as math;

/// Рисует УСЛОВИЕ задачи (то, что видит ребёнок). Правильного ответа тут нет —
/// он живёт только на сервере в active_tasks.answer.
class TaskVisual extends StatelessWidget {
  const TaskVisual({super.key, required this.payload});
  final Map<String, dynamic> payload;

  int _i(dynamic v, [int d = 0]) => (v as num?)?.toInt() ?? d;
  String _s(dynamic v) => v?.toString() ?? '';

  @override
  Widget build(BuildContext context) {
    // ИСПРАВЛЕНО: читаем kind из payload, если есть; иначе определяем по наличию полей
    final kind = _s(payload['kind']);
    
    if (kind.isNotEmpty) {
      switch (kind) {
        case 'mul':
          return _Equation('${_i(payload['a'])} × ${_i(payload['b'])}');
        case 'arith':
          final op = _s(payload['op']) == 'sub' ? '−' : '+';
          return _Equation('${_i(payload['a'])} $op ${_i(payload['b'])}');
        case 'read_clock':
          return _ClockFace(hour: _i(payload['hour']), minute: _i(payload['minute']));
        case 'compare_frac':
          return _CompareFractions(
            leftNum: _i(payload['left_num']),
            rightNum: _i(payload['right_num']),
            den: _i(payload['den'], 4),
          );
        case 'compose':
          return _Equation('Досыпь до ${_i(payload['total'])} (уже ${_i(payload['given'])})');
        case 'shade_frac':
          return _Equation('Закрась ${_i(payload['need'])} из ${_i(payload['parts'])}');
        default:
          break; // падаем в fallback ниже
      }
    }
    
    // FALLBACK: если kind отсутствует или неизвестен — определяем тип по структуре данных
    if (payload.containsKey('a') && payload.containsKey('b')) {
      // Это арифметика (add/sub/mul)
      if (payload.containsKey('options')) {
        return _Equation('${_i(payload['a'])} × ${_i(payload['b'])}');
      } else {
        final op = _s(payload['op']) == 'sub' ? '−' : '+';
        return _Equation('${_i(payload['a'])} $op ${_i(payload['b'])}');
      }
    } else if (payload.containsKey('hour') && payload.containsKey('minute')) {
      return _ClockFace(hour: _i(payload['hour']), minute: _i(payload['minute']));
    } else if (payload.containsKey('left_num') && payload.containsKey('den')) {
      return _CompareFractions(
        leftNum: _i(payload['left_num']),
        rightNum: _i(payload['right_num']),
        den: _i(payload['den'], 4),
      );
    } else if (payload.containsKey('total') && payload.containsKey('given')) {
      return _Equation('Досыпь до ${_i(payload['total'])} (уже ${_i(payload['given'])})');
    } else if (payload.containsKey('parts') && payload.containsKey('need')) {
      return _Equation('Закрась ${_i(payload['need'])} из ${_i(payload['parts'])}');
    }
    
    // Если ничего не подошло — показываем сырые данные для диагностики
    return Padding(
      padding: const EdgeInsets.all(12),
      child: Text(
        'DEBUG: keys=${payload.keys.join(', ')}, values=${payload.values.map((v) => v.toString()).join(', ')}',
        style: const TextStyle(fontSize: 12, color: Colors.orange),
      ),
    );
  }
}

class _ClockPainter extends CustomPainter {
  _ClockPainter({required this.hour, required this.minute});
  final int hour, minute;

  @override
  void paint(Canvas canvas, Size size) {
    final c = Offset(size.width / 2, size.height / 2);
    final r = size.width / 2 - 6;
    canvas.drawCircle(
        c,
        r,
        Paint()
          ..style = PaintingStyle.stroke
          ..strokeWidth = 4
          ..color = AppColors.purpleLight);
    final tick = Paint()
      ..color = Colors.white70
      ..strokeWidth = 3;
    for (var i = 0; i < 12; i++) {
      final a = (math.pi / 6) * i;
      canvas.drawLine(
        c + Offset(math.sin(a), -math.cos(a)) * (r - 10),
        c + Offset(math.sin(a), -math.cos(a)) * r,
        tick,
      );
    }
    final hAng = (math.pi / 6) * ((hour % 12) + minute / 60);
    final mAng = (math.pi / 30) * minute;
    canvas.drawLine(
      c,
      c + Offset(math.sin(hAng), -math.cos(hAng)) * (r * 0.5),
      Paint()
        ..color = Colors.white
        ..strokeWidth = 6
        ..strokeCap = StrokeCap.round,
    );
    canvas.drawLine(
      c,
      c + Offset(math.sin(mAng), -math.cos(mAng)) * (r * 0.78),
      Paint()
        ..color = AppColors.gold
        ..strokeWidth = 4
        ..strokeCap = StrokeCap.round,
    );
    canvas.drawCircle(c, 6, Paint()..color = AppColors.gold);
  }

  @override
  bool shouldRepaint(covariant _ClockPainter old) =>
      old.hour != hour || old.minute != minute;
}
