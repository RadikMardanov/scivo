/// Фон с градиентом ночи + мерцающие звёзды. Оборачивает любой контент.
class MagicBackground extends StatelessWidget {
  const MagicBackground({super.key, required this.child});
  final Widget child;

  @override
  Widget build(BuildContext context) {
    return Stack(
      children: [
        Positioned.fill(
          child: DecoratedBox(
            decoration: BoxDecoration(gradient: AppTheme.nightGradient),
          ),
        ),
        Positioned.fill(
          child: IgnorePointer(
            child: CustomPaint(painter: _StarsPainter()),
          ),
        ),
        SafeArea(child: child),
      ],
    );
  }
}

class _StarsPainter extends CustomPainter {
  @override
  void paint(Canvas canvas, Size size) {
    final p = Paint()..color = Colors.white.withValues(alpha: 0.5);
    // Псевдослучайная раскладка звёзд (детерминированная, без Random)
    for (var i = 0; i < 40; i++) {
      final x = ((i * 97) % 100) / 100 * size.width;
      final y = ((i * 53) % 100) / 100 * size.height;
      final r = (i % 3 == 0) ? 1.8 : 1.0;
      canvas.drawCircle(Offset(x, y), r, p);
    }
  }

  @override
  bool shouldRepaint(covariant _StarsPainter old) => false;
}

/// Каркас экрана: фон + optional AppBar + body.

    super.key,
    required this.body,
    this.title,
    this.actions,
    this.bottomNavigationBar,
  });

  final Widget body;
  final String? title;
  final List<Widget>? actions;
  final Widget? bottomNavigationBar;

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      extendBodyBehindAppBar: true,
      backgroundColor: Colors.transparent,
      appBar: title == null
          ? null
          : AppBar(
              backgroundColor: Colors.transparent,
              elevation: 0,
              centerTitle: true,
              title: Text(
                title!,
                style: const TextStyle(
                    color: Colors.white, fontWeight: FontWeight.w900, fontSize: 20),
              ),
              actions: actions,
            ),
      body: MagicBackground(child: body),
      bottomNavigationBar: bottomNavigationBar,
    );
  }
}

/// Диалог Scivo (заменяет AlertDialog для единообразия).
Future<void> showScivoDialog(BuildContext context,
    {required String text, String button = 'Ок'}) {
  return showDialog<void>(
    context: context,
    builder: (_) => AlertDialog(
      backgroundColor: AppColors.nightCard,
      shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(20)),
      content: Text(text,
          style: const TextStyle(color: Colors.white, fontSize: 16, height: 1.4)),
      actions: [
        TextButton(
            onPressed: () => Navigator.pop(context),
            child: Text(button,
                style: const TextStyle(
                    color: AppColors.gold, fontWeight: FontWeight.w800))),
      ],
    ),
  );
}

/// Русская плюрализация: pluralRu(5, 'герой', 'героя', 'героев').
String pluralRu(int n, String one, String few, String many) {
  final mod10 = n % 10;
  final mod100 = n % 100;
  if (mod10 == 1 && mod100 != 11) return one;
  if (mod10 >= 2 && mod10 <= 4 && !(mod100 >= 12 && mod100 <= 14)) return few;
  return many;
}

/// Маршрутизатор верхнего уровня: решает, какой экран показать в зависимости
/// от состояния авторизации, согласия родителя и наличия активного героя.
class AuthGate extends ConsumerWidget {
  const AuthGate({super.key});

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    final session = ref.watch(authSessionProvider);
    final parentAsync = ref.watch(parentProvider);
    final childrenAsync = ref.watch(childrenProvider);
    final activeKid = ref.watch(activeChildProvider);

    // Не авторизован → экран входа/регистрации.
    if (session == null) return const AuthScreen();

    // Авторизован, но согласие ещё не дано → экран согласия.
    final parent = parentAsync.valueOrNull;
    if (parent != null && !parent.hasConsent) return const ConsentScreen();

    // Согласие есть, но детей ещё нет → экран добавления первого героя.
    final kids = childrenAsync.valueOrNull ?? const [];
    if (kids.isEmpty) return const ProfilePicker(isFirstTime: true);

    // Дети есть, активный выбран → домашний шелтер ребёнка.
    if (activeKid != null) return const ChildShell();

    // Дети есть, но никто не выбран → пикер героев.
    return const ProfilePicker();
  }
}

/// Вкладка вишлиста: список желаний от родителя с эмодзи, названием и ценой.
class WishlistTab extends ConsumerWidget {
  const WishlistTab({super.key, required this.kid, required this.progress});
  final ChildProfile kid;
  final ChildProgress progress;

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    final items = ref.watch(wishlistProvider).valueOrNull ?? const <WishlistItem>[];

    if (items.isEmpty) {
      return Center(
        child: Padding(
          padding: const EdgeInsets.all(24),
          child: MagicCard(
            child: Column(
              mainAxisSize: MainAxisSize.min,
              children: [
                const Icon(Icons.card_giftcard_rounded,
                    size: 64, color: AppColors.purpleLight),
                const SizedBox(height: 12),
                Text('Вишлист пуст 🎁\nПопроси маму или папу добавить подарок!',
                    textAlign: TextAlign.center,
                    style: const TextStyle(fontSize: 16, height: 1.4)),
              ],
            ),
          ),
        ),
      );
    }

    return ListView.separated(
      padding: const EdgeInsets.fromLTRB(16, 8, 16, 24),
      itemCount: items.length,
      separatorBuilder: (_, __) => const SizedBox(height: 12),
      itemBuilder: (_, i) {
        final w = items[i];
        final enough = progress.feathers >= w.featherCost;
        return MagicCard(
          child: Row(
            children: [
              Text(w.emoji, style: const TextStyle(fontSize: 40)),
              const SizedBox(width: 12),
              Expanded(
                child: Column(
                  crossAxisAlignment: CrossAxisAlignment.start,
                  children: [
                    Text(w.title,
                        style: const TextStyle(
                            fontSize: 18, fontWeight: FontWeight.w800)),
                    const SizedBox(height: 4),
                    Text('${w.featherCost} 🪶',
                        style: TextStyle(
                            fontSize: 16,
                            fontWeight: FontWeight.w700,
                            color: enough ? AppColors.gold : Colors.white54)),
                  ],
                ),
              ),
              FilledButton.tonal(
                onPressed: enough ? null : () {},
                child: Text(enough ? 'Есть!' : 'Хочу'),
              ),
            ],
          ),
        );
      },
    );
  }
}
