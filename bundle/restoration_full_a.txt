import 'package:supabase_flutter/supabase_flutter.dart';

/// Конфигурация окружения. Секреты SUPABASE_URL и SUPABASE_ANON_KEY
/// передаются через --dart-define при сборке в GitHub Actions.
class EnvConfig {
  static late final SupabaseClient supabase;

  static Future<void> initSupabase() async {
    const url = String.fromEnvironment('SUPABASE_URL');
    const anonKey = String.fromEnvironment('SUPABASE_ANON_KEY');
    
    if (url.isEmpty || anonKey.isEmpty) {
      throw Exception('Missing SUPABASE_URL or SUPABASE_ANON_KEY. Use --dart-define.');
    }
    
    await Supabase.initialize(url: url, anonKey: anonKey);
    supabase = Supabase.instance.client;
  }
}

import 'dart:convert';

import 'package:supabase_flutter/supabase_flutter.dart';

import '../core/errors.dart';
import '../data/tasks.dart';
import '../models/family_models.dart';

class SupabaseService {
  SupabaseService(this._client);

  final SupabaseClient _client;

  Session? get currentSession => _client.auth.currentSession;
  Stream<AuthState> get authChanges => _client.auth.onAuthStateChange;

  // ---------- Аккаунт родителя ----------
  Future<void> signUpParent({
    required String email,
    required String password,
    required String displayName,
  }) async {
    await _client.auth.signUp(
      email: email.trim(),
      password: password,
      data: {
        'display_name': displayName.trim(),
        'consent': 'true',
      },
    );
  }

  Future<void> signIn(String email, String password) async {
    await _client.auth.signInWithPassword(
      email: email.trim(),
      password: password,
    );
  }

  Future<void> signOut() => _client.auth.signOut();

  Future<ParentProfile?> fetchParent() async {
    final userId = _client.auth.currentUser?.id;
    if (userId == null) return null;
    final row = await _client
        .from('parents')
        .select('id, display_name, has_pin, consent_at')
        .eq('id', userId)
        .maybeSingle();
    return row == null ? null : ParentProfile.fromJson(row);
  }

  Future<void> giveConsent() async {
    await _client.rpc('give_consent');
  }

  Future<void> setParentPin(String pin) async {
    final result =
        await _client.rpc('set_parent_pin', params: {'p_new_pin': pin});
    final check = PinCheckResult.fromJson(_asMap(result));
    if (!check.ok) throw AppException(check.message);
  }

  Future<PinCheckResult> verifyParentPin(String pin) async {
    final result =
        await _client.rpc('verify_parent_pin', params: {'p_pin': pin});
    return PinCheckResult.fromJson(_asMap(result));
  }

  // ---------- Дети ----------
  Future<List<ChildProfile>> fetchChildren() async {
    final rows = await _client
        .from('children')
        .select(
            'id, nickname, avatar_id, grade, onboarding_done, daily_feather_cap')
        .order('created_at');
    return rows.map((row) => ChildProfile.fromJson(row)).toList();
  }

  Future<String> createChild({
    required String nickname,
    required String avatarId,
    required int grade,
    required String pin,
  }) async {
    final result = await _client.rpc('create_child', params: {
      'p_nickname': nickname.trim(),
      'p_avatar_id': avatarId,
      'p_grade': grade,
      'p_pin': pin,
    });
    return result.toString();
  }

  Future<PinCheckResult> verifyChildPin(String childId, String pin) async {
    final result = await _client.rpc('verify_child_pin', params: {
      'p_child_id': childId,
      'p_pin': pin,
    });
    return PinCheckResult.fromJson(_asMap(result));
  }

  // ---------- Прогресс ----------
  Future<ChildProgress> fetchProgress(String childId) async {
    final row = await _client
        .from('child_progress')
        .select()
        .eq('child_id', childId)
        .maybeSingle();
    return row == null ? ChildProgress.empty : ChildProgress.fromJson(row);
  }

  Future<CheckinResult> checkin(String childId) async {
    final raw =
        await _client.rpc('child_checkin', params: {'p_child_id': childId});
    final result = _asMap(raw);
    return CheckinResult(
      progress:
          ChildProgress.fromJson(_asMap(result['progress'] as Map? ?? const {})),
      manaBonus: (result['mana_bonus'] as num?)?.toInt() ?? 0,
      usedFreeze: result['used_freeze'] == true,
    );
  }

  Future<int> completeOnboarding(String childId) async {
    final raw = await _client
        .rpc('complete_onboarding', params: {'p_child_id': childId});
    final result = _asMap(raw);
    return (result['mana_bonus'] as num?)?.toInt() ?? 0;
  }

  // ---------- Вишлист ----------
  Future<List<WishlistItem>> fetchWishlist() async {
    final rows = await _client
        .from('wishlist_items')
        .select('id, title, feather_cost, emoji')
        .eq('is_active', true)
        .order('feather_cost');
    return rows.map((row) => WishlistItem.fromJson(row)).toList();
  }

  // ---------- Задания (миграция 002) ----------
  /// Выдаёт новую задачу по миру. Ответ на клиент не приходит.
  /// HOTFIX 002e: явно сериализуем params в JSON через jsonEncode/jsonDecode,
  /// чтобы supabase_flutter гарантированно отправил их серверу как объект.
  Future<GeneratedTask> generateTask(String childId, String worldId) async {
    final paramsJson = jsonEncode({
      'p_child_id': childId,
      'p_world_id': worldId,
    });
    final params = jsonDecode(paramsJson) as Map<String, dynamic>;
    final raw = await _client.rpc('generate_task', params: params);
    return GeneratedTask.fromJson(_asMap(raw));
  }

  /// Отправляет ответ на сервер; сервер решает, верно ли, и начисляет награды.
  Future<SubmitResult> submitAnswer(
    String childId,
    String taskId,
    Map<String, dynamic> input,
  ) async {
    final paramsJson = jsonEncode({
      'p_child_id': childId,
      'p_task_id': taskId,
      'p_input': input,
    });
    final params = jsonDecode(paramsJson) as Map<String, dynamic>;
    final raw = await _client.rpc('submit_answer', params: params);
    return SubmitResult.fromJson(_asMap(raw));
  }

  /// Просит подсказку уровня 1..3 (тратит ману на сервере).
  Future<HintResult> requestHint(String childId, String taskId, int level) async {

      'p_child_id': childId,
      'p_task_id': taskId,
      'p_level': level,

    final raw = await _client.rpc('request_hint', params: params);
    return HintResult.fromJson(_asMap(raw));
  }

  Map<String, dynamic> _asMap(dynamic value) =>
      Map<String, dynamic>.from(value as Map? ?? const {});
}

/// SVG-ассеты воронов Scivo (6 цветов аватара).
/// Кэшируем строки, чтобы не парсить каждый кадр.
class RavenArt {
  static final Map<String, String> _cache = {};

  static String svg(String avatarId) {
    if (_cache.containsKey(avatarId)) return _cache[avatarId]!;
    
    // Здесь должны быть реальные SVG-строки для каждого аватара.
    // Для демонстрации используем placeholder.
    const placeholders = {
      'raven_blue': '<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 100 100"><circle cx="50" cy="50" r="40" fill="#4A90D9"/></svg>',
      'raven_green': '<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 100 100"><circle cx="50" cy="50" r="40" fill="#4CAF50"/></svg>',
      'raven_red': '<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 100 100"><circle cx="50" cy="50" r="40" fill="#F44336"/></svg>',
      'raven_yellow': '<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 100 100"><circle cx="50" cy="50" r="40" fill="#FFEB3B"/></svg>',
      'raven_purple': '<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 100 100"><circle cx="50" cy="50" r="40" fill="#9C27B0"/></svg>',
      'raven_black': '<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 100 100"><circle cx="50" cy="50" r="40" fill="#212121"/></svg>',

    final result = placeholders[avatarId] ?? placeholders['raven_blue']!;

class _StarsPainter extends CustomPainter {
  @override
  void paint(Canvas canvas, Size size) {
    final p = Paint()..color = Colors.white.withValues(alpha: 0.5);
    // Псевдослучайная раскладка звёзд (детерминированная, без Random)
    for (var i = 0; i < 40; i++) {
      final x = ((i * 97) % 100) / 100 * size.width;
      final y = ((i * 53) % 100) / 100 * size.height;
      final r = (i % 3 == 0) ? 1.8 : 1.0;
      canvas.drawCircle(Offset(x, y), r, p);
    }
  }

  @override
  bool shouldRepaint(covariant _StarsPainter old) => false;
}

/// Русская плюрализация: pluralRu(5, 'герой', 'героя', 'героев').
String pluralRu(int n, String one, String few, String many) {
  final mod10 = n % 10;
  final mod100 = n % 100;
  if (mod10 == 1 && mod100 != 11) return one;
  if (mod10 >= 2 && mod10 <= 4 && !(mod100 >= 12 && mod100 <= 14)) return few;
  return many;
}

/// Анимированный воронёнок Scivo (покачивается и моргает).
class ScivoRaven extends StatefulWidget {
  const ScivoRaven({super.key, required this.size, this.avatarId = 'raven_blue', this.animated = true});
  final double size;
  final String avatarId;
  final bool animated;

  @override
  State<ScivoRaven> createState() => _ScivoRavenState();
}

class _ScivoRavenState extends State<ScivoRaven> with SingleTickerProviderStateMixin {
  late AnimationController _controller;
  Timer? _blinkTimer;
  bool _blinking = false;

  @override
  void initState() {
    super.initState();
    _controller = AnimationController(vsync: this, duration: const Duration(seconds: 2))..repeat(reverse: true);
    if (widget.animated) {
      _startBlinking();
    }
  }

  void _startBlinking() {
    _blinkTimer = Timer.periodic(const Duration(milliseconds: 3000), (_) {
      setState(() => _blinking = true);
      Future.delayed(const Duration(milliseconds: 150), () {
        if (mounted) setState(() => _blinking = false);
      });
    });
  }

  @override
  void dispose() {
    _controller.dispose();
    _blinkTimer?.cancel();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return AnimatedBuilder(
      animation: _controller,
      builder: (_, __) {
        final angle = (_controller.value - 0.5) * 0.1; // лёгкое покачивание
        return Transform.rotate(
          angle: widget.animated ? angle : 0,
          child: SvgPicture.string(
            RavenArt.svg(widget.avatarId),
            width: widget.size,
            height: widget.size,
          ),
        );
      },
    );
  }
}
