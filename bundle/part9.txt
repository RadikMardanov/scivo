/// Экран квеста: выдаёт задачу с сервера, принимает ответ, показывает
/// подсказки и награды. Правильность решает ТОЛЬКО сервер (RPC 002).
class QuestScreen extends ConsumerStatefulWidget {
  const QuestScreen({
    super.key,
    required this.childId,
    required this.worldId,
    required this.worldTitle,
  });

  final String childId;
  final String worldId;
  final String worldTitle;

  @override
  ConsumerState<QuestScreen> createState() => _QuestScreenState();
}

class _QuestScreenState extends ConsumerState<QuestScreen> {
  @override
  void initState() {
    super.initState();
    WidgetsBinding.instance.addPostFrameCallback((_) => _load());
  }

  @override
  void dispose() {
    ref.read(questControllerProvider.notifier).reset();
    super.dispose();
  }

  void _load() =>
      ref.read(questControllerProvider.notifier).loadTask(widget.childId, widget.worldId);

  void _submit(Map<String, dynamic> input) =>
      ref.read(questControllerProvider.notifier).submit(widget.childId, input);

  void _hint() {
    final level = ref.read(questControllerProvider).hintLevel;
    ref.read(questControllerProvider.notifier).askHint(widget.childId, level + 1);
  }

  @override
  Widget build(BuildContext context) {
    final state = ref.watch(questControllerProvider);
    final task = state.task;
    final solved = state.lastResult?.correct == true;
    final avatarId = ref.watch(activeChildProvider)?.avatarId ?? 'raven_blue';

    Widget body;
    if (task == null) {
      body = state.error != null
          ? ErrorView(message: state.error!, onRetry: _load)
          : const Center(child: CircularProgressIndicator());
    } else {
      final supported =
          task.mechanic == Mechanics.choice || task.mechanic == Mechanics.keypad;
      body = Column(
        children: [
          Expanded(
            child: SingleChildScrollView(
              padding: const EdgeInsets.fromLTRB(20, 8, 20, 8),
              child: Column(
                children: [
                  ScivoRaven(size: 96, avatarId: avatarId),
                  const SizedBox(height: 8),
                  SpeechBubble(_prompt(task.payload)),
                  const SizedBox(height: 16),
                  TaskVisual(payload: task.payload),
                  if (state.lastHint != null) ...[
                    const SizedBox(height: 12),
                    _HintBubble(state.lastHint!.text),
                  ],
                  if (state.lastResult != null && !solved) ...[
                    const SizedBox(height: 12),
                    const Text(
                      'Почти! Посмотри ещё раз — я рядом. 🐦‍⬛',
                      textAlign: TextAlign.center,
                      style: TextStyle(color: Colors.white70),
                    ),
                  ],
                ],
              ),
            ),
          ),
          Padding(
            padding: const EdgeInsets.fromLTRB(20, 0, 20, 16),
            child: solved
                ? _SolvedPanel(result: state.lastResult!, onNext: _load)
                : !supported
                    ? _UnsupportedPanel(onAnother: _load)
                    : Column(
                        children: [
                          _InputFor(
                            task: task,
                            enabled: !state.isLoading,
                            onSubmit: _submit,
                          ),
                          const SizedBox(height: 12),
                          SizedBox(
                            width: double.infinity,
                            child: OutlinedButton.icon(
                              onPressed:
                                  (state.isLoading || state.hintLevel >= 3) ? null : _hint,
                              icon: const Icon(Icons.lightbulb_outline_rounded),
                              label: Text('Подсказка · 5 ✨ (${state.hintLevel}/3)'),
                            ),
                          ),
                        ],
                      ),
          ),
        ],
      );
    }

    return MagicScaffold(title: widget.worldTitle, body: body);
  }

  String _prompt(Map<String, dynamic> p) {
    switch (p['kind']?.toString() ?? '') {
      case 'mul':
        return 'Кар! Умножай быстро — дракон ждёт!';
      case 'arith':
        return p['op'] == 'sub' ? 'Вычитай без страха!' : 'Складывай смелее!';
      case 'read_clock':
        return 'Посмотри на стрелки — который час?';
      case 'compare_frac':
        return 'Какая дробь больше?';
      case 'compose':
        return 'Досыпь жёлуди до кучи!';
      case 'shade_frac':
        return 'Закрась нужные дольки!';
      default:
        return 'Реши задачу, я в тебя верю!';
    }
  }
}

class _InputFor extends StatelessWidget {
  const _InputFor({required this.task, required this.enabled, required this.onSubmit});

  final GeneratedTask task;
  final bool enabled;
  final void Function(Map<String, dynamic>) onSubmit;

  @override
  Widget build(BuildContext context) {
    if (task.mechanic == Mechanics.choice) {
      final opts =
          (task.payload['options'] as List?)?.cast<String>() ?? const <String>[];
      return ChoiceInput(options: opts, enabled: enabled, onSubmit: onSubmit);
    }
    return KeypadInput(enabled: enabled, onSubmit: onSubmit);
  }
}

import 'dart:math' as math;

import 'package:flutter/material.dart';

import '../../core/theme/app_theme.dart';

/// Рисует УСЛОВИЕ задачи (то, что видит ребёнок). Правильного ответа тут нет —
/// он живёт только на сервере в active_tasks.answer.
class TaskVisual extends StatelessWidget {
  const TaskVisual({super.key, required this.payload});
  final Map<String, dynamic> payload;

  int _i(dynamic v, [int d = 0]) => (v as num?)?.toInt() ?? d;
  String _s(dynamic v) => v?.toString() ?? '';

  @override
  Widget build(BuildContext context) {
    switch (_s(payload['kind'])) {
      case 'mul':
        return _Equation('${_i(payload['a'])} × ${_i(payload['b'])}');
      case 'arith':
        final op = _s(payload['op']) == 'sub' ? '−' : '+';
        return _Equation('${_i(payload['a'])} $op ${_i(payload['b'])}');
      case 'read_clock':
        return _ClockFace(hour: _i(payload['hour']), minute: _i(payload['minute']));
      case 'compare_frac':
        return _CompareFractions(
          leftNum: _i(payload['left_num']),
          rightNum: _i(payload['right_num']),
          den: _i(payload['den'], 4),
        );
      case 'compose':
        return _Equation('Досыпь до ${_i(payload['total'])} (уже ${_i(payload['given'])})');
      case 'shade_frac':
        return _Equation('Закрась ${_i(payload['need'])} из ${_i(payload['parts'])}');
      default:
        return const SizedBox.shrink();
    }
  }
}

class _ClockFace extends StatelessWidget {
  const _ClockFace({required this.hour, required this.minute});
  final int hour, minute;

  @override
  Widget build(BuildContext context) => Center(
        child: SizedBox(

          height: 180,
          child: CustomPaint(painter: _ClockPainter(hour: hour, minute: minute)),

class _ClockPainter extends CustomPainter {
  _ClockPainter({required this.hour, required this.minute});
  final int hour, minute;

  @override
  void paint(Canvas canvas, Size size) {
    final c = Offset(size.width / 2, size.height / 2);
    final r = size.width / 2 - 6;
    canvas.drawCircle(
        c,
        r,
        Paint()
          ..style = PaintingStyle.stroke
          ..strokeWidth = 4
          ..color = AppColors.purpleLight);
    final tick = Paint()
      ..color = Colors.white70
      ..strokeWidth = 3;
    for (var i = 0; i < 12; i++) {
      final a = (math.pi / 6) * i;
      canvas.drawLine(
        c + Offset(math.sin(a), -math.cos(a)) * (r - 10),
        c + Offset(math.sin(a), -math.cos(a)) * r,
        tick,
      );
    }
    final hAng = (math.pi / 6) * ((hour % 12) + minute / 60);
    final mAng = (math.pi / 30) * minute;
    canvas.drawLine(
      c,
      c + Offset(math.sin(hAng), -math.cos(hAng)) * (r * 0.5),
      Paint()
        ..color = Colors.white
        ..strokeWidth = 6
        ..strokeCap = StrokeCap.round,
    );
    canvas.drawLine(
      c,
      c + Offset(math.sin(mAng), -math.cos(mAng)) * (r * 0.78),
      Paint()
        ..color = AppColors.gold
        ..strokeWidth = 4
        ..strokeCap = StrokeCap.round,
    );
    canvas.drawCircle(c, 6, Paint()..color = AppColors.gold);
  }

  @override
  bool shouldRepaint(covariant _ClockPainter old) =>
      old.hour != hour || old.minute != minute;
}

import 'package:flutter/material.dart';

import '../../../core/theme/app_theme.dart';

/// Механика «числовая клавиатура»: крупные кнопки 0–9, дробная черта «/»,
/// стереть, «Ответ». Работает и для целых (8+5), и для дробей (3/4).
class KeypadInput extends StatefulWidget {
  const KeypadInput({super.key, required this.enabled, required this.onSubmit});

  final bool enabled;
  final void Function(Map<String, dynamic>) onSubmit;

  @override
  State<KeypadInput> createState() => _KeypadInputState();
}

class _KeypadInputState extends State<KeypadInput> {
  String _text = '';

  void _tap(String d) {
    if (!widget.enabled) return;
    setState(() {
      if (_text.length < 8) _text += d;
    });
  }

  void _slash() {
    if (!widget.enabled) return;
    setState(() {
      if (_text.length < 8 && !_text.contains('/')) _text += '/';
    });
  }

  void _back() {
    if (!widget.enabled || _text.isEmpty) return;
    setState(() => _text = _text.substring(0, _text.length - 1));
  }

  void _send() {
    if (!widget.enabled || _text.trim().isEmpty) return;
    widget.onSubmit({'value': _text.trim()});
  }

  @override
  Widget build(BuildContext context) {
    const rows = [
      ['1', '2', '3'],
      ['4', '5', '6'],
      ['7', '8', '9'],
    ];
    return Column(
      children: [
        Container(
          width: double.infinity,
          alignment: Alignment.center,
          padding: const EdgeInsets.symmetric(vertical: 14),
          decoration: BoxDecoration(
            color: Colors.black.withValues(alpha: 0.25),
            borderRadius: BorderRadius.circular(16),
            border: Border.all(color: AppColors.gold.withValues(alpha: 0.4)),
          ),
          child: Text(
            _text.isEmpty ? '?' : _text,
            style: const TextStyle(
                fontSize: 34,
                fontWeight: FontWeight.w900,
                color: AppColors.gold,
                letterSpacing: 2),
          ),
        ),
        const SizedBox(height: 12),
        for (final row in rows)
          Padding(
            padding: const EdgeInsets.only(bottom: 8),
            child: Row(
              mainAxisAlignment: MainAxisAlignment.center,
              children: [
                for (final k in row) _Key(label: k, onTap: () => _tap(k)),
              ],
            ),
          ),
        Padding(
          padding: const EdgeInsets.only(bottom: 8),
          child: Row(
            mainAxisAlignment: MainAxisAlignment.center,
            children: [
              _Key(label: '/', onTap: _slash),
              _Key(label: '0', onTap: () => _tap('0')),
              _Key(label: '⌫', onTap: _back),
            ],
          ),
        ),
        const SizedBox(height: 4),
        SizedBox(
          width: double.infinity,
          child: FilledButton(
            onPressed: widget.enabled ? _send : null,
            child: const Text('Ответ ✨'),
          ),
        ),
      ],
    );
  }
}

}
