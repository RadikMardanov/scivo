// --- Сервис ---
final serviceProvider = Provider<SupabaseService>((ref) {
  return SupabaseService(EnvConfig.supabase);
});

// --- Сессия аутентификации ---
final authSessionProvider = StreamProvider<Session?>((ref) {
  final service = ref.watch(serviceProvider);
  return service.authChanges.map((state) => state.session);
});

// --- Профиль родителя ---
final parentProvider = FutureProvider<ParentProfile?>((ref) async {
  final session = ref.watch(authSessionProvider).valueOrNull;
  if (session == null) return null;
  final service = ref.read(serviceProvider);
  return service.fetchParent();
});

// --- Список детей ---
final childrenProvider = FutureProvider<List<ChildProfile>>((ref) async {
  final session = ref.watch(authSessionProvider).valueOrNull;
  if (session == null) return [];
  final service = ref.read(serviceProvider);
  return service.fetchChildren();
});

// --- Активный ребёнок (выбранный в picker) ---
final activeChildProvider = StateProvider<ChildProfile?>((ref) => null);

// --- Прогресс конкретного ребёнка ---

    FutureProvider.family<ChildProgress, String>((ref, childId) async {
  final service = ref.read(serviceProvider);
  return service.fetchProgress(childId);

// --- Вишлист ---

class _ChildCard extends ConsumerStatefulWidget {
  const _ChildCard({required this.child});
  final ChildProfile child;

  @override
  // ИСПРАВЛЕНО: createState обязан вернуть ConsumerState<_ChildCard>,
  // т.к. родительский класс — ConsumerStatefulWidget.
  ConsumerState<_ChildCard> createState() => _ChildCardState();
}

class _ChildCardState extends ConsumerState<_ChildCard> {
  bool _showPin = false;
  String _pin = '';
  bool _isLoading = false;

  void _onDigit(String d) {
    if (_pin.length < 4) setState(() => _pin += d);
  }

  void _onDelete() {
    if (_pin.isNotEmpty) setState(() => _pin = _pin.substring(0, _pin.length - 1));
  }

  Future<void> _verifyAndEnter() async {
    if (_pin.length != 4) return;

    setState(() => _isLoading = true);
    try {
      final service = ref.read(serviceProvider);
      final result = await service.verifyChildPin(widget.child.id, _pin);
      if (result.ok) {
        ref.read(activeChildProvider.notifier).state = widget.child;
        // ИСПРАВЛЕНО: pushReplacement в Flutter 3.47 требует два type argumentа.
        Navigator.pushReplacement<void, void>(
          context,
          MaterialPageRoute(builder: (_) => const ChildShell()),
        );
      } else {
        showSnack(context, result.message, isError: true);
        setState(() => _pin = '');
      }
    } catch (e) {
      showSnack(context, e.toString(), isError: true);
    } finally {
      if (mounted) setState(() => _isLoading = false);
    }
  }

  @override
  Widget build(BuildContext context) {
    if (_showPin) {
      return MagicCard(
        child: Column(
          children: [
            Text('Введите PIN для ${widget.child.nickname}', textAlign: TextAlign.center, style: Theme.of(context).textTheme.titleMedium),
            const SizedBox(height: 16),
            Row(
              mainAxisAlignment: MainAxisAlignment.center,
              children: List.generate(4, (i) => Container(
                margin: const EdgeInsets.symmetric(horizontal: 8),
                width: 40,
                height: 40,
                decoration: BoxDecoration(
                  color: i < _pin.length ? Colors.white : Colors.white.withValues(alpha: 0.2),
                  shape: BoxShape.circle,
                ),
              )),
            ),
            const SizedBox(height: 24),
            PinPad(
              onDigit: _onDigit,
              onDelete: _onDelete,
              onOk: _verifyAndEnter,
              enabled: !_isLoading,
            ),
            const SizedBox(height: 16),
            OutlinedButton(
              onPressed: () => setState(() => _showPin = false),
              child: const Text('Отмена'),
            ),
          ],
        ),
      );
    }

    return Card(
      elevation: 4,
      shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(16)),
      child: ListTile(
        leading: CircleAvatar(
          backgroundColor: AppColors.purpleLight,
          child: Text(widget.child.nickname.substring(0, 1).toUpperCase(), style: const TextStyle(fontWeight: FontWeight.w900)),
        ),
        title: Text(widget.child.nickname, style: const TextStyle(fontWeight: FontWeight.w800, fontSize: 18)),
        subtitle: Text('${widget.child.grade} класс'),
        trailing: const Icon(Icons.arrow_forward_ios_rounded),
        onTap: () => setState(() => _showPin = true),
      ),
    );
  }
}

/// Экран входа/регистрации родителя.
class AuthScreen extends ConsumerStatefulWidget {
  const AuthScreen({super.key});

  @override
  ConsumerState<AuthScreen> createState() => _AuthScreenState();
}

class _AuthScreenState extends ConsumerState<AuthScreen> {
  final _formKey = GlobalKey<FormState>();
  final _emailController = TextEditingController();
  final _passwordController = TextEditingController();
  final _displayNameController = TextEditingController();
  bool _isLogin = true;
  bool _isLoading = false;

  @override
  void dispose() {
    _emailController.dispose();
    _passwordController.dispose();
    _displayNameController.dispose();
    super.dispose();
  }

  Future<void> _submit() async {
    if (!_formKey.currentState!.validate()) return;

    setState(() => _isLoading = true);
    try {
      final service = ref.read(serviceProvider);
      if (_isLogin) {
        await service.signIn(_emailController.text, _passwordController.text);
      } else {
        await service.signUpParent(
          email: _emailController.text,
          password: _passwordController.text,
          displayName: _displayNameController.text,
        );
      }
    } catch (e) {
      if (mounted) showSnack(context, friendlyError(e), isError: true);
    } finally {
      if (mounted) setState(() => _isLoading = false);
    }
  }

  @override
  Widget build(BuildContext context) {
    return MagicScaffold(
      title: _isLogin ? 'Вход в магию' : 'Новый герой',
      body: Form(
        key: _formKey,
        child: SingleChildScrollView(
          padding: const EdgeInsets.all(20),
          child: Column(
            crossAxisAlignment: CrossAxisAlignment.stretch,
            children: [
              Center(child: ScivoRaven(size: 120, avatarId: 'raven_blue')),
              const SizedBox(height: 16),
              Text(
                _isLogin ? 'Рады видеть тебя снова!' : 'Создай аккаунт родителя',
                textAlign: TextAlign.center,
                style: Theme.of(context).textTheme.headlineSmall,
              ),
              const SizedBox(height: 24),
              if (!_isLogin) ...[
                TextFormField(
                  controller: _displayNameController,
                  validator: (v) => (v == null || v.trim().isEmpty) ? 'Введите имя' : null,
                  decoration: magicInput('Как вас зовут?', icon: Icons.person_outline_rounded),
                ),
                const SizedBox(height: 16),
              ],
              TextFormField(
                controller: _emailController,
                keyboardType: TextInputType.emailAddress,
                validator: (v) {
                  if (v == null || v.isEmpty) return 'Введите email';
                  if (!v.contains('@')) return 'Некорректный email';
                  return null;
                },
                decoration: magicInput('Email родителя', icon: Icons.email_rounded),
              ),
              const SizedBox(height: 16),
              TextFormField(
                controller: _passwordController,
                obscureText: true,
                validator: (v) {
                  if (v == null || v.isEmpty) return 'Введите пароль';
                  if (v.length < 6) return 'Минимум 6 символов';
                  return null;
                },
                decoration: magicInput('Пароль', icon: Icons.lock_outline_rounded),
              ),
              const SizedBox(height: 32),
              FilledButton.icon(
                onPressed: _isLoading ? null : _submit,
                icon: _isLoading
                    ? const SizedBox(width: 18, height: 18, child: CircularProgressIndicator(strokeWidth: 2))
                    // ИСПРАВЛЕНО: Icons.magic_button_rounded не существует,
                    // заменяем на существующую auto_awesome_rounded.
                    : const Icon(Icons.auto_awesome_rounded),
                label: Text(_isLoading ? 'Магия...' : (_isLogin ? 'Войти' : 'Зарегистрироваться')),
              ),
              const SizedBox(height: 16),
              TextButton(
                onPressed: _isLoading ? null : () => setState(() => _isLogin = !_isLogin),
                child: Text(
                  _isLogin ? 'Нет аккаунта? Создать' : 'Уже есть аккаунт? Войти',
                  style: const TextStyle(color: AppColors.gold, fontWeight: FontWeight.w700),
                ),
              ),
            ],
          ),
        ),
      ),
    );
  }
}

/// Экран создания нового героя (ребёнка).
class AddChildScreen extends ConsumerStatefulWidget {
  const AddChildScreen({super.key, this.isFirst = false});
  final bool isFirst;

  @override
  ConsumerState<AddChildScreen> createState() => _AddChildScreenState();
}

class _AddChildScreenState extends ConsumerState<AddChildScreen> {
  final _nameController = TextEditingController();
  String _selectedAvatar = 'raven_blue';
  int _selectedGrade = 1;
  String _pin = '';
  int _step = 0; // 0=name/avatar, 1=grade, 2=pin
  bool _isLoading = false;

  static const _avatars = ['raven_blue', 'raven_green', 'raven_red', 'raven_yellow', 'raven_purple', 'raven_black'];
  static const _grades = [1, 2, 3, 4];

  @override
  void dispose() {
    _nameController.dispose();
    super.dispose();
  }

  void _nextStep() {
    if (_step == 0) {
      if (_nameController.text.trim().isEmpty) {
        showSnack(context, 'Введите имя героя', isError: true);
        return;
      }
      setState(() => _step = 1);
    } else if (_step == 1) {
      setState(() => _step = 2);
    }
  }

  void _prevStep() {
    if (_step > 0) setState(() => _step--);
  }

  void _onDigit(String d) {
    if (_pin.length < 4) setState(() => _pin += d);
  }

  void _onDelete() {
    if (_pin.isNotEmpty) setState(() => _pin = _pin.substring(0, _pin.length - 1));
  }

  Future<void> _finish() async {
    if (_pin.length != 4) {
      showSnack(context, 'PIN должен состоять из 4 цифр', isError: true);
      return;
    }

    setState(() => _isLoading = true);
    try {
      final service = ref.read(serviceProvider);
      await service.createChild(
        nickname: _nameController.text.trim(),
        avatarId: _selectedAvatar,
        grade: _selectedGrade,
        pin: _pin,
      );
      ref.invalidate(childrenProvider);
      if (mounted) {
        // ИСПРАВЛЕНО: $_name не существовал, читаем значение из контроллера.
        final name = _nameController.text.trim();
        showSnack(context, 'Герой $name готов к приключениям! 🎉');
        Navigator.pop(context, true);
      }
    } catch (e) {
      if (mounted) showSnack(context, friendlyError(e), isError: true);
    } finally {
      if (mounted) setState(() => _isLoading = false);
    }
  }

  @override
  Widget build(BuildContext context) {
    return MagicScaffold(
      title: 'Новый герой',
      showBack: !widget.isFirst,
      body: SingleChildScrollView(
        padding: const EdgeInsets.all(20),
        child: Column(
          crossAxisAlignment: CrossAxisAlignment.stretch,
          children: [
            Center(child: ScivoRaven(size: 80, avatarId: _selectedAvatar)),
            const SizedBox(height: 16),

            if (_step == 0) ...[
              Text('Как звать героя?', textAlign: TextAlign.center, style: Theme.of(context).textTheme.headlineSmall),
              const SizedBox(height: 16),
              TextField(
                controller: _nameController,
                maxLength: 20,
                decoration: magicInput('Имя героя', icon: Icons.face_retouching_natural_rounded),
              ),
              const SizedBox(height: 24),
              Text('Выберите цвет оперения:', style: Theme.of(context).textTheme.titleMedium),
              const SizedBox(height: 12),
              Wrap(
                spacing: 12,
                runSpacing: 12,
                alignment: WrapAlignment.center,
                children: _avatars.map((a) => GestureDetector(
                  onTap: () => setState(() => _selectedAvatar = a),
                  child: Container(
                    padding: const EdgeInsets.all(8),
                    decoration: BoxDecoration(
                      color: _selectedAvatar == a ? AppColors.gold : Colors.white.withValues(alpha: 0.08),
                      borderRadius: BorderRadius.circular(12),
                      border: Border.all(color: _selectedAvatar == a ? AppColors.textDark : Colors.transparent, width: 2),
                    ),
                    child: ScivoRaven(size: 48, avatarId: a, animated: false),
                  ),
                )).toList(),
              ),
              const SizedBox(height: 32),
              FilledButton(onPressed: _nextStep, child: const Text('Далее')),
            ] else if (_step == 1) ...[
              Text('Какой класс?', textAlign: TextAlign.center, style: Theme.of(context).textTheme.headlineSmall),
              const SizedBox(height: 16),
              Wrap(
                spacing: 12,
                runSpacing: 12,
                alignment: WrapAlignment.center,
                children: _grades.map((g) => ChoiceChip(
                  label: Text('$g класс'),
                  selected: _selectedGrade == g,
                  onSelected: (_) => setState(() => _selectedGrade = g),
                  selectedColor: AppColors.gold,
                  backgroundColor: Colors.white.withValues(alpha: 0.1),
                )).toList(),
              ),
              const SizedBox(height: 32),
              Row(
                children: [
                  Expanded(child: OutlinedButton(onPressed: _prevStep, child: const Text('Назад'))),
                  const SizedBox(width: 16),
                  Expanded(child: FilledButton(onPressed: _nextStep, child: const Text('Далее'))),
                ],
              ),
            ] else ...[
              Text('Придумайте PIN (4 цифры)', textAlign: TextAlign.center, style: Theme.of(context).textTheme.headlineSmall),
              const SizedBox(height: 16),
              Row(
                mainAxisAlignment: MainAxisAlignment.center,
                children: List.generate(4, (i) => Container(
                  margin: const EdgeInsets.symmetric(horizontal: 8),
                  width: 40,
                  height: 40,

                    color: i < _pin.length ? Colors.white : Colors.white.withValues(alpha: 0.2),
                    shape: BoxShape.circle,
