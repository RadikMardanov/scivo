/// Решает, какой экран показать, в зависимости от состояния семьи.
class AuthGate extends ConsumerWidget {
  const AuthGate({super.key});

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    ref.listen<AsyncValue<Session?>>(authSessionProvider, (previous, next) {
      if (next.valueOrNull == null) {
        ref.read(activeChildProvider.notifier).state = null;
      }
    });

    final session = ref.watch(authSessionProvider);
    return session.when(
      loading: () => const SplashView(),
      error: (error, _) => _GateError(
        message: friendlyError(error),
        onRetry: () => ref.invalidate(authSessionProvider),
      ),
      data: (value) => value == null ? const AuthScreen() : const _SignedInGate(),
    );
  }
}

class _SignedInGate extends ConsumerWidget {
  const _SignedInGate();

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    final parentAsync = ref.watch(parentProvider);

    return parentAsync.when(
      loading: () => const SplashView(),
      error: (error, _) => _GateError(
        message: friendlyError(error),
        onRetry: () => ref.invalidate(parentProvider),
      ),
      data: (parent) {
        if (parent == null) {
          return _GateError(
            message: 'Профиль родителя не найден. Выйдите и войдите снова.',
            onRetry: () => ref.invalidate(parentProvider),
          );
        }
        if (!parent.hasConsent) return const ConsentScreen();
        if (!parent.hasPin) return const ParentPinSetupScreen();

        final childrenAsync = ref.watch(childrenProvider);
        return childrenAsync.when(
          loading: () => const SplashView(),
          error: (error, _) => _GateError(
            message: friendlyError(error),
            onRetry: () => ref.invalidate(childrenProvider),
          ),
          data: (kids) {
            if (kids.isEmpty) return const AddChildScreen(isFirst: true);
            final active = ref.watch(activeChildProvider);
            if (active != null) return ChildShell(kid: active);
            return ProfilePickerScreen(kids: kids);
          },
        );
      },
    );
  }
}

class _GateError extends ConsumerWidget {
  const _GateError({required this.message, required this.onRetry});

  final String message;
  final VoidCallback onRetry;

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    return Scaffold(
      body: MagicBackground(
        child: SafeArea(
          child: ErrorView(
            message: message,
            onRetry: onRetry,
            extra: TextButton(
              onPressed: () => ref.read(serviceProvider).signOut(),
              child: const Text('Выйти из аккаунта'),
            ),
          ),
        ),
      ),
    );
  }
}

class _AuthScreenState extends ConsumerState<AuthScreen> {
  final _name = TextEditingController();
  final _email = TextEditingController();
  final _password = TextEditingController();

  bool _isRegister = false;
  bool _consent = false;
  bool _loading = false;
  bool _showPassword = false;
  String? _error;

  @override
  void dispose() {
    _name.dispose();
    _email.dispose();
    _password.dispose();
    super.dispose();
  }

  Future<void> _submit() async {
    final email = _email.text.trim();
    final password = _password.text;

    String? problem;
    if (!email.contains('@') || !email.contains('.')) {
      problem = 'Введите корректный email.';
    } else if (password.length < 6) {
      problem = 'Пароль — минимум 6 символов.';
    } else if (_isRegister && !_consent) {
      problem = 'Отметьте согласие на обработку данных.';
    }
    if (problem != null) {
      setState(() => _error = problem);
      return;
    }

    setState(() {
      _loading = true;
      _error = null;
    });

    final service = ref.read(serviceProvider);
    try {
      if (_isRegister) {
        await service.signUpParent(
          email: email,
          password: password,
          displayName: _name.text,
        );
        if (service.currentSession == null && mounted) {
          setState(() => _error =
              'Аккаунт создан. Подтвердите email по ссылке из письма и войдите.');
        }
      } else {
        await service.signIn(email, password);
      }
    } catch (error) {
      if (mounted) setState(() => _error = friendlyError(error));
    } finally {
      if (mounted) setState(() => _loading = false);
    }
  }

        actions: [
          TextButton(
            onPressed: () => Navigator.of(dialogContext).pop(),
            child: const Text('Закрыть'),
          ),
        ],
      ),
    );
  }

                  ],

/// Родительский замок: пример на умножение + PIN родителя (проверка на сервере).
class ParentGateScreen extends ConsumerStatefulWidget {
  const ParentGateScreen({super.key});

  @override
  ConsumerState<ParentGateScreen> createState() => _ParentGateScreenState();
}

class _ParentGateScreenState extends ConsumerState<ParentGateScreen> {
  final _random = math.Random();
  final _answer = TextEditingController();
  int _a = 7;
  int _b = 8;
  bool _mathPassed = false;
  String? _message;

  @override
  void initState() {
    super.initState();
    _newTask();
  }

  @override
  void dispose() {
    _answer.dispose();
    super.dispose();
  }

  void _newTask() {
    _a = 6 + _random.nextInt(4);
    _b = 6 + _random.nextInt(4);
    _answer.clear();
  }

  void _checkMath() {
    if (int.tryParse(_answer.text.trim()) == _a * _b) {
      setState(() {
        _mathPassed = true;
        _message = null;
      });
    } else {
      setState(() {
        _message = 'Неверно. Вот новый пример.';
        _newTask();
      });
    }
  }

  Future<bool> _onPin(String pin) async {
    try {
      final result = await ref.read(serviceProvider).verifyParentPin(pin);
      if (!mounted) return false;
      if (result.ok) {
        Navigator.of(context).pushReplacement(
          MaterialPageRoute<void>(builder: (_) => const ParentZoneScreen()),
        );
        return true;
      }
      setState(() => _message = result.message);
      return false;
    } catch (error) {
      if (mounted) setState(() => _message = friendlyError(error));
      return false;
    }
  }

  @override
  Widget build(BuildContext context) {
    final text = Theme.of(context).textTheme;
    final message = _message;

    return MagicScaffold(
      title: 'Для взрослых',
      body: Center(
        child: SingleChildScrollView(
          padding: const EdgeInsets.all(24),
          child: ConstrainedBox(
            constraints: const BoxConstraints(maxWidth: 420),
            child: Column(
              mainAxisSize: MainAxisSize.min,
              children: [
                if (!_mathPassed) ...[
                  const Text('🔒', style: TextStyle(fontSize: 48)),
                  const SizedBox(height: 8),
                  Text('Этот раздел для родителей', style: text.titleLarge),
                  const SizedBox(height: 16),
                  Text(
                    'Сколько будет $_a × $_b?',
                    style: text.headlineSmall?.copyWith(
                      color: AppColors.gold,
                      fontWeight: FontWeight.w800,
                    ),
                  ),
                  const SizedBox(height: 16),
                  TextField(
                    controller: _answer,
                    keyboardType: TextInputType.number,
                    textAlign: TextAlign.center,
                    style: const TextStyle(fontSize: 28, fontWeight: FontWeight.w800),
                    decoration: magicInput('Ответ'),
                    onSubmitted: (_) => _checkMath(),
                  ),
                  const SizedBox(height: 16),
                  SizedBox(
                    width: double.infinity,
                    child: FilledButton(
                      onPressed: _checkMath,
                      child: const Text('Продолжить'),
                    ),
                  ),
                ] else ...[
                  Text('Введите PIN родителя', style: text.titleLarge),
                  const SizedBox(height: 16),
                  PinPad(onCompleted: _onPin),
                ],
                if (message != null)
                  Padding(
                    padding: const EdgeInsets.only(top: 16),
                    child: Text(
                      message,
                      textAlign: TextAlign.center,
                      style: const TextStyle(
                        color: AppColors.danger,
                        fontWeight: FontWeight.w700,
                      ),
                    ),
                  ),
              ],
            ),
          ),
        ),
      ),
    );
  }
