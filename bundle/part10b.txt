import 'package:flutter/material.dart';

import '../../../core/theme/app_theme.dart';

/// Механика «перетаскивание»: ребёнок перетаскивает предметы в корзины.
/// Для part10 реализована как тапы по ячейкам (упрощённая drag-and-drop).
class DragInput extends StatefulWidget {
  const DragInput({super.key, required this.total, required this.given, required this.enabled, required this.onSubmit});

  final int total;
  final int given;
  final bool enabled;
  final void Function(Map<String, dynamic>) onSubmit;

  @override
  State<DragInput> createState() => _DragInputState();
}

class _DragInputState extends State<DragInput> {
  late int _rightCount;

  @override
  void initState() {
    super.initState();
    // ИСПРАВЛЕНО: начинаем с нуля, а не с правильного ответа
    _rightCount = 0;
  }

  void _tapRight(int delta) {
    if (!widget.enabled) return;
    setState(() {
      _rightCount = (_rightCount + delta).clamp(0, widget.total - widget.given);
    });
  }

  void _send() {
    if (!widget.enabled) return;
    widget.onSubmit({'right': _rightCount});
  }

  @override
  Widget build(BuildContext context) {
    return Column(
      children: [
        Row(
          mainAxisAlignment: MainAxisAlignment.spaceEvenly,
          children: [
            _Basket(label: 'Левая (${widget.given})', count: widget.given, color: AppColors.purpleLight),
            _Basket(label: 'Правая ($_rightCount)', count: _rightCount, color: AppColors.gold),
          ],
        ),
        const SizedBox(height: 16),
        Row(
          mainAxisAlignment: MainAxisAlignment.center,
          children: [
            IconButton(onPressed: () => _tapRight(-1), icon: const Icon(Icons.remove_circle_outline_rounded)),
            const SizedBox(width: 16),
            TextButton(onPressed: _send, child: const Text('Готово ✨')),
            const SizedBox(width: 16),
            IconButton(onPressed: () => _tapRight(1), icon: const Icon(Icons.add_circle_outline_rounded)),
          ],
        ),
      ],
    );
  }
}

class _Basket extends StatelessWidget {
  const _Basket({required this.label, required this.count, required this.color});
  final String label;
  final int count;
  final Color color;

  @override
  Widget build(BuildContext context) => Container(

          color: color.withValues(alpha: 0.2),

            Text(label, style: const TextStyle(fontWeight: FontWeight.w700)),
            Text('$count', style: const TextStyle(fontSize: 32, fontWeight: FontWeight.w900)),
          ],
        ),
      );
}

import 'package:flutter/material.dart';

import '../../../core/theme/app_theme.dart';

/// Механика «закрашивание долей»: пицца/круг делится на части, ребёнок тапает нужные.
class ShadeInput extends StatefulWidget {
  const ShadeInput({super.key, required this.parts, required this.need, required this.enabled, required this.onSubmit});

  final int parts;
  final int need;
  final bool enabled;
  final void Function(Map<String, dynamic>) onSubmit;

  @override
  State<ShadeInput> createState() => _ShadeInputState();
}

class _ShadeInputState extends State<ShadeInput> {
  late Set<int> _shaded;

  @override
  void initState() {
    super.initState();
    _shaded = {};
  }

  void _toggle(int index) {
    if (!widget.enabled) return;
    setState(() {
      if (_shaded.contains(index)) {
        _shaded.remove(index);
      } else if (_shaded.length < widget.need) {
        _shaded.add(index);
      }
    });
  }

  void _send() {
    if (!widget.enabled) return;
    widget.onSubmit({'shaded': _shaded.toList()..sort()});
  }

  @override
  Widget build(BuildContext context) {
    final p = widget.parts.clamp(1, 12);
    return Column(
      children: [
        SizedBox(
          height: MediaQuery.sizeOf(context).width * 0.8, // адаптивная высота
          child: GridView.builder(
            physics: const NeverScrollableScrollPhysics(),
            gridDelegate: SliverGridDelegateWithFixedCrossAxisCount(
              crossAxisCount: p <= 4 ? 2 : 3,
              mainAxisSpacing: 8,
              crossAxisSpacing: 8,
            ),
            itemCount: p,
            itemBuilder: (_, i) => GestureDetector(
              onTap: () => _toggle(i),
              child: AnimatedContainer(
                duration: const Duration(milliseconds: 200),
                decoration: BoxDecoration(
                  color: _shaded.contains(i) ? AppColors.gold : Colors.white.withValues(alpha: 0.1),
                  borderRadius: BorderRadius.circular(12),
                  border: Border.all(color: _shaded.contains(i) ? AppColors.textDark : Colors.white24, width: 2),
                ),
                alignment: Alignment.center,
                child: Text('${i + 1}', style: TextStyle(
                  fontSize: 24,
                  fontWeight: FontWeight.w900,
                  color: _shaded.contains(i) ? AppColors.textDark : Colors.white70,
                )),
              ),
            ),
          ),
        ),
        const SizedBox(height: 12),
        Text('Закрась ${widget.need} из $p', textAlign: TextAlign.center),
        const SizedBox(height: 12),
        FilledButton(onPressed: _send, child: const Text('Ответ ✨')),
      ],
    );
  }
}

/// Экран квеста: выдаёт задачу с сервера, принимает ответ, показывает
/// подсказки и награды. Правильность решает ТОЛЬКО сервер (RPC 002).
class QuestScreen extends ConsumerStatefulWidget {
  const QuestScreen({
    super.key,
    required this.childId,
    required this.worldId,
    required this.worldTitle,
  });

  final String childId;
  final String worldId;
  final String worldTitle;

  @override
  ConsumerState<QuestScreen> createState() => _QuestScreenState();
}

class _QuestScreenState extends ConsumerState<QuestScreen> {
  @override
  void initState() {
    super.initState();
    WidgetsBinding.instance.addPostFrameCallback((_) => _load());
  }

  @override
  void dispose() {
    ref.read(questControllerProvider.notifier).reset();
    super.dispose();
  }

  void _load() =>
      ref.read(questControllerProvider.notifier).loadTask(widget.childId, widget.worldId);

  void _submit(Map<String, dynamic> input) =>
      ref.read(questControllerProvider.notifier).submit(widget.childId, input);

  void _hint() {
    final level = ref.read(questControllerProvider).hintLevel;
    ref.read(questControllerProvider.notifier).askHint(widget.childId, level + 1);
  }

  @override
  Widget build(BuildContext context) {
    final state = ref.watch(questControllerProvider);
    final task = state.task;
    final solved = state.lastResult?.correct == true;
    final avatarId = ref.watch(activeChildProvider)?.avatarId ?? 'raven_blue';

    Widget body;
    if (task == null) {
      body = state.error != null
          ? ErrorView(message: state.error!, onRetry: _load)
          : const Center(child: CircularProgressIndicator());
    } else {
      body = Column(
        children: [
          Expanded(
            child: SingleChildScrollView(
              padding: const EdgeInsets.fromLTRB(20, 8, 20, 8),
              child: Column(
                children: [
                  ScivoRaven(size: 96, avatarId: avatarId),
                  const SizedBox(height: 8),
                  SpeechBubble(_prompt(task.payload)),
                  const SizedBox(height: 16),
                  TaskVisual(payload: task.payload),
                  if (state.lastHint != null) ...[
                    const SizedBox(height: 12),
                    _HintBubble(state.lastHint!.text),
                  ],
                  if (state.lastResult != null && !solved) ...[
                    const SizedBox(height: 12),
                    const Text(
                      'Почти! Посмотри ещё раз — я рядом. 🐦‍',
                      textAlign: TextAlign.center,
                      style: TextStyle(color: Colors.white70),
                    ),
                  ],
                ],
              ),
            ),
          ),
          Padding(
            padding: const EdgeInsets.fromLTRB(20, 0, 20, 16),
            child: solved
                ? _SolvedPanel(result: state.lastResult!, onNext: _load)
                : Column(
                    children: [
                      _InputFor(
                        task: task,
                        enabled: !state.isLoading,
                        onSubmit: _submit,
                      ),
                      const SizedBox(height: 12),
                      SizedBox(
                        width: double.infinity,
                        child: OutlinedButton.icon(
                          onPressed:
                              (state.isLoading || state.hintLevel >= 3) ? null : _hint,
                          icon: const Icon(Icons.lightbulb_outline_rounded),
                          label: Text('Подсказка · 5 ✨ (${state.hintLevel}/3)'),
                        ),
                      ),
                    ],
                  ),
          ),
        ],
      );
    }

    return MagicScaffold(title: widget.worldTitle, body: body);
  }

  String _prompt(Map<String, dynamic> p) {
    switch (p['kind']?.toString() ?? '') {
      case 'mul':
        return 'Кар! Умножай быстро — дракон ждёт!';
      case 'arith':
        return p['op'] == 'sub' ? 'Вычитай без страха!' : 'Складывай смелее!';
      case 'read_clock':
        return 'Посмотри на стрелки — который час?';
      case 'compare_frac':
        return 'Какая дробь больше?';
      case 'compose':
        return 'Досыпь жёлуди до кучи!';
      case 'shade_frac':
        return 'Закрась нужные дольки!';
      default:
        return 'Реши задачу, я в тебя верю!';
    }
  }
}

class _InputFor extends StatelessWidget {
  const _InputFor({required this.task, required this.enabled, required this.onSubmit});

  final GeneratedTask task;
  final bool enabled;
  final void Function(Map<String, dynamic>) onSubmit;

  @override
  Widget build(BuildContext context) {
    final kind = task.payload['kind']?.toString() ?? '';
    switch (kind) {
      case 'compose':
        return DragInput(
          total: (task.payload['total'] as num?)?.toInt() ?? 10,
          given: (task.payload['given'] as num?)?.toInt() ?? 0,
          enabled: enabled,
          onSubmit: onSubmit,
        );
      case 'shade_frac':
        return ShadeInput(
          parts: (task.payload['parts'] as num?)?.toInt() ?? 8,
          need: (task.payload['need'] as num?)?.toInt() ?? 1,
          enabled: enabled,
          onSubmit: onSubmit,
        );
      case 'mul':
      case 'read_clock':
      case 'compare_frac':
        final opts = (task.payload['options'] as List?)?.cast<String>() ?? const <String>[];
        return ChoiceInput(options: opts, enabled: enabled, onSubmit: onSubmit);
      default: // arith
        return KeypadInput(enabled: enabled, onSubmit: onSubmit);
    }
  }
}

class _SolvedPanel extends StatelessWidget {
  const _SolvedPanel({required this.result, required this.onNext});
  final SubmitResult result;
  final VoidCallback onNext;

  @override
  Widget build(BuildContext context) {
    final r = result.reward;
    final parts = <String>[];
    if (r.feather > 0) parts.add('+${r.feather} ');
    if (r.mana > 0) parts.add('+${r.mana} ✨');
    if (r.xp > 0) parts.add('+${r.xp} ⭐');
    final flavor = result.feedbackKey == 'success_clean'
        ? 'Без подсказок — красавчик!'
        : 'Умница, подсказка помогла!';
    return MagicCard(
      child: Column(
        children: [
          Text(
            '🎉 Верно! ${parts.join('  ')}',
            textAlign: TextAlign.center,
            style: const TextStyle(
                fontSize: 20, fontWeight: FontWeight.w900, color: AppColors.gold),
          ),
          const SizedBox(height: 6),
          Text(
            '$flavor${r.fixBonus > 0 ? '  Ты исправил ошибку — молодец!' : ''}',
            textAlign: TextAlign.center,
            style: const TextStyle(color: Colors.white70),
          ),
          const SizedBox(height: 16),
          SizedBox(
            width: double.infinity,
            child: FilledButton(
                onPressed: onNext, child: const Text('Следующая задача')),
          ),
        ],
      ),
    );
  }
}

/// Карта 5 миров. Все острова активны после part10.
class WorldMapTab extends StatelessWidget {
  const WorldMapTab({super.key, required this.kid});
  final ChildProfile kid;

  bool _playable(WorldInfo w) => w.fitsGrade(kid.grade);

  void _open(BuildContext context, WorldInfo w) {
    if (_playable(w)) {
      Navigator.of(context).push(MaterialPageRoute<void>(
        builder: (_) =>
            QuestScreen(childId: kid.id, worldId: w.id, worldTitle: w.title),
      ));
    } else {
      showScivoDialog(
        context,
        text: '${w.emoji} ${w.title}\n\n'
            'Этот остров пока недоступен для твоего класса. Попробуй другой!',
        button: 'Жду!',
      );
    }
  }

  @override
  Widget build(BuildContext context) {
    return ListView(
      padding: const EdgeInsets.fromLTRB(16, 8, 16, 24),
      children: [
        Center(child: SpeechBubble('Куда полетим сегодня, ${kid.nickname}?')),
        const SizedBox(height: 16),
        for (var i = 0; i < kWorlds.length; i++) ...[
          if (i > 0) const _PathDots(),
          Align(
            alignment: i.isEven ? Alignment.centerLeft : Alignment.centerRight,
            child: FractionallySizedBox(
              widthFactor: 0.88,
              child: _WorldCard(
                world: kWorlds[i],
                recommended: kWorlds[i].fitsGrade(kid.grade),
                playable: _playable(kWorlds[i]),
                onTap: () => _open(context, kWorlds[i]),
              ),
            ),
          ),
        ],
      ],
    );
  }
}
