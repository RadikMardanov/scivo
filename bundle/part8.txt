/// Задача, выданная сервером (generate_task).
class GeneratedTask {
  const GeneratedTask({
    required this.taskId,
    required this.worldId,
    required this.mechanic,
    required this.difficulty,
    required this.payload,
  });

  final String taskId;
  final String worldId;
  final String mechanic;
  final int difficulty;
  final Map<String, dynamic> payload;

  factory GeneratedTask.fromJson(Map<String, dynamic> json) {
    return GeneratedTask(
      taskId: json['task_id'].toString(),
      worldId: json['world_id'].toString(),
      mechanic: json['mechanic'].toString(),
      difficulty: (json['difficulty'] as num?)?.toInt() ?? 1,
      payload: Map<String, dynamic>.from(json['payload'] as Map? ?? const {}),
    );
  }
}

/// Награда за попытку (submit_answer -> reward).
class Reward {
  const Reward({
    required this.feather,
    required this.mana,
    required this.xp,
    required this.fixBonus,
  });

  final int feather;
  final int mana;
  final int xp;
  final int fixBonus;

  static const zero = Reward(feather: 0, mana: 0, xp: 0, fixBonus: 0);

  factory Reward.fromJson(Map<String, dynamic> json) {
    return Reward(
      feather: (json['feather'] as num?)?.toInt() ?? 0,
      mana: (json['mana'] as num?)?.toInt() ?? 0,
      xp: (json['xp'] as num?)?.toInt() ?? 0,
      fixBonus: (json['fix_bonus'] as num?)?.toInt() ?? 0,
    );
  }
}

/// Результат проверки ответа сервером (submit_answer).
class SubmitResult {
  const SubmitResult({
    required this.correct,
    required this.reward,
    required this.feedbackKey,
    required this.hintLevel,
  });

  final bool correct;
  final Reward reward;
  final String feedbackKey; // success_clean | success_hinted | wrong
  final int hintLevel;

  factory SubmitResult.fromJson(Map<String, dynamic> json) {
    return SubmitResult(
      correct: json['correct'] == true,
      reward: Reward.fromJson(
        Map<String, dynamic>.from(json['reward'] as Map? ?? const {}),
      ),
      feedbackKey: json['feedback_key']?.toString() ?? 'wrong',
      hintLevel: (json['hint_level'] as num?)?.toInt() ?? 0,
    );
  }
}

/// Результат запроса подсказки (request_hint).
class HintResult {
  const HintResult({
    required this.text,
    required this.level,
    required this.manaSpent,
  });

  final String text;
  final int level;
  final int manaSpent;

  factory HintResult.fromJson(Map<String, dynamic> json) {
    final hint = json['hint'];
    final text = hint is Map
        ? (hint['text']?.toString() ?? '')
        : (hint?.toString() ?? '');
    return HintResult(
      text: text.isEmpty
          ? 'Подсказка готовится. Попробуй ещё раз, я рядом!'
          : text,
      level: (json['level'] as num?)?.toInt() ?? 0,
      manaSpent: (json['mana_spent'] as num?)?.toInt() ?? 0,
    );
  }
}

import 'package:supabase_flutter/supabase_flutter.dart';

import '../core/errors.dart';
import '../data/tasks.dart';
import '../models/family_models.dart';

class SupabaseService {
  SupabaseService(this._client);

  final SupabaseClient _client;

  Session? get currentSession => _client.auth.currentSession;
  Stream<AuthState> get authChanges => _client.auth.onAuthStateChange;

  // ---------- Аккаунт родителя ----------
  Future<void> signUpParent({
    required String email,
    required String password,
    required String displayName,
  }) async {
    await _client.auth.signUp(
      email: email.trim(),
      password: password,
      data: {
        'display_name': displayName.trim(),
        'consent': 'true',
      },
    );
  }

  Future<void> signIn(String email, String password) async {
    await _client.auth.signInWithPassword(
      email: email.trim(),
      password: password,
    );
  }

  Future<void> signOut() => _client.auth.signOut();

  Future<ParentProfile?> fetchParent() async {
    final userId = _client.auth.currentUser?.id;
    if (userId == null) return null;
    final row = await _client
        .from('parents')
        .select('id, display_name, has_pin, consent_at')
        .eq('id', userId)
        .maybeSingle();
    return row == null ? null : ParentProfile.fromJson(row);
  }

  Future<void> giveConsent() async {
    await _client.rpc('give_consent');
  }

  Future<void> setParentPin(String pin) async {
    final result =
        await _client.rpc('set_parent_pin', params: {'p_new_pin': pin});
    final check = PinCheckResult.fromJson(_asMap(result));
    if (!check.ok) throw AppException(check.message);
  }

  Future<PinCheckResult> verifyParentPin(String pin) async {
    final result =
        await _client.rpc('verify_parent_pin', params: {'p_pin': pin});
    return PinCheckResult.fromJson(_asMap(result));
  }

  // ---------- Дети ----------
  Future<List<ChildProfile>> fetchChildren() async {
    final rows = await _client
        .from('children')
        .select(
            'id, nickname, avatar_id, grade, onboarding_done, daily_feather_cap')
        .order('created_at');
    return rows.map((row) => ChildProfile.fromJson(row)).toList();
  }

  Future<String> createChild({
    required String nickname,
    required String avatarId,
    required int grade,
    required String pin,
  }) async {
    final result = await _client.rpc('create_child', params: {
      'p_nickname': nickname.trim(),
      'p_avatar_id': avatarId,
      'p_grade': grade,
      'p_pin': pin,
    });
    return result.toString();
  }

  Future<PinCheckResult> verifyChildPin(String childId, String pin) async {
    final result = await _client.rpc('verify_child_pin', params: {
      'p_child_id': childId,
      'p_pin': pin,
    });
    return PinCheckResult.fromJson(_asMap(result));
  }

  // ---------- Прогресс ----------
  Future<ChildProgress> fetchProgress(String childId) async {
    final row = await _client
        .from('child_progress')
        .select()
        .eq('child_id', childId)
        .maybeSingle();
    return row == null ? ChildProgress.empty : ChildProgress.fromJson(row);
  }

  Future<CheckinResult> checkin(String childId) async {
    final raw =
        await _client.rpc('child_checkin', params: {'p_child_id': childId});
    final result = _asMap(raw);
    return CheckinResult(
      progress:
          ChildProgress.fromJson(_asMap(result['progress'] as Map? ?? const {})),
      manaBonus: (result['mana_bonus'] as num?)?.toInt() ?? 0,
      usedFreeze: result['used_freeze'] == true,
    );
  }

  Future<int> completeOnboarding(String childId) async {
    final raw = await _client
        .rpc('complete_onboarding', params: {'p_child_id': childId});
    final result = _asMap(raw);
    return (result['mana_bonus'] as num?)?.toInt() ?? 0;
  }

  // ---------- Вишлист ----------
  Future<List<WishlistItem>> fetchWishlist() async {
    final rows = await _client
        .from('wishlist_items')
        .select('id, title, feather_cost, emoji')
        .eq('is_active', true)
        .order('feather_cost');
    return rows.map((row) => WishlistItem.fromJson(row)).toList();
  }

  // ---------- Задания (миграция 002) ----------
  /// Выдаёт новую задачу по миру. Ответ на клиент не приходит.
  Future<GeneratedTask> generateTask(String childId, String worldId) async {
    final raw = await _client.rpc('generate_task', params: {
      'p_child_id': childId,
      'p_world_id': worldId,
    });
    return GeneratedTask.fromJson(_asMap(raw));
  }

  /// Отправляет ответ на сервер; сервер решает, верно ли, и начисляет награды.
  Future<SubmitResult> submitAnswer(
    String childId,
    String taskId,
    Map<String, dynamic> input,
  ) async {
    final raw = await _client.rpc('submit_answer', params: {
      'p_child_id': childId,
      'p_task_id': taskId,
      'p_input': input,
    });
    return SubmitResult.fromJson(_asMap(raw));
  }

  /// Просит подсказку уровня 1..3 (тратит ману на сервере).
  Future<HintResult> requestHint(String childId, String taskId, int level) async {
    final raw = await _client.rpc('request_hint', params: {
      'p_child_id': childId,
      'p_task_id': taskId,
      'p_level': level,
    });
    return HintResult.fromJson(_asMap(raw));
  }

  Map<String, dynamic> _asMap(dynamic value) =>
      Map<String, dynamic>.from(value as Map? ?? const {});
}

import 'package:flutter_riverpod/flutter_riverpod.dart';

import '../core/errors.dart';
import '../data/tasks.dart';

import 'app_providers.dart';

/// Состояние активного квеста: одна задача + последняя проверка + подсказки.
class QuestState {
  const QuestState({
    this.task,
    this.isLoading = false,
    this.error,
    this.lastResult,
    this.hintLevel = 0,
    this.lastHint,
  });

  final GeneratedTask? task;
  final bool isLoading;
  final String? error;
  final SubmitResult? lastResult;
  final int hintLevel; // сколько подсказок уже открыто по текущей задаче
  final HintResult? lastHint;
}

class QuestController extends StateNotifier<QuestState> {
  QuestController(this._ref) : super(const QuestState());

  final Ref _ref;

  SupabaseService get _service => _ref.read(serviceProvider);

  /// Загружает новую задачу по миру (сбрасывает предыдущее состояние).
  Future<void> loadTask(String childId, String worldId) async {
    state = const QuestState(isLoading: true);
    try {
      final task = await _service.generateTask(childId, worldId);
      state = QuestState(task: task);
    } catch (error) {
      state = QuestState(error: friendlyError(error));
    }
  }

  /// Отправляет ответ; при успехе обновляет прогресс (перья/мана в шапке).
  Future<void> submit(String childId, Map<String, dynamic> input) async {
    final task = state.task;
    if (task == null) return;
    state = state.copyWith(isLoading: true, clearError: true);
    try {
      final result = await _service.submitAnswer(childId, task.taskId, input);
      state = state.copyWith(isLoading: false, lastResult: result);
      _ref.invalidate(progressProvider(childId));
    } catch (error) {
      state = state.copyWith(isLoading: false, error: friendlyError(error));
    }
  }

  /// Запрашивает подсказку уровня 1..3 (один раз на уровень).
  Future<void> askHint(String childId, int level) async {
    final task = state.task;
    if (task == null) return;
    if (level <= state.hintLevel) return; // этот уровень уже давали
    try {
      final hint = await _service.requestHint(childId, task.taskId, level);
      state = state.copyWith(lastHint: hint, hintLevel: level);
      _ref.invalidate(progressProvider(childId)); // мана списалась
    } catch (error) {
      state = state.copyWith(error: friendlyError(error));
    }
  }

  void reset() => state = const QuestState();
}

final questControllerProvider =
    StateNotifierProvider<QuestController, QuestState>((ref) {
  return QuestController(ref);
});

extension on QuestState {
  QuestState copyWith({
    bool? isLoading,
    String? error,
    SubmitResult? lastResult,
    HintResult? lastHint,
    int? hintLevel,
    bool clearError = false,
  }) {
    return QuestState(
      task: task,
      isLoading: isLoading ?? this.isLoading,
      error: clearError ? null : (error ?? this.error),
      lastResult: lastResult ?? this.lastResult,
      hintLevel: hintLevel ?? this.hintLevel,
      lastHint: lastHint ?? this.lastHint,
    );
  }
}
