/// Создание профиля ребёнка: имя -> класс -> цвет вороненка -> PIN (2 раза).
class AddChildScreen extends ConsumerStatefulWidget {
  const AddChildScreen({super.key, this.isFirst = false});

  final bool isFirst;

  @override
  ConsumerState<AddChildScreen> createState() => _AddChildScreenState();
}

class _AddChildScreenState extends ConsumerState<AddChildScreen> {
  final _name = TextEditingController();
  int _step = 0;
  int? _grade;
  String _avatar = RavenArt.avatars.first.id;
  String? _pin;
  String? _message;

  @override
  void dispose() {
    _name.dispose();
    super.dispose();
  }

  void _toPin() {
    if (_name.text.trim().isEmpty) {
      setState(() => _message = 'Введите имя или никнейм героя.');
      return;
    }
    if (_grade == null) {
      setState(() => _message = 'Выберите класс.');
      return;
    }
    setState(() {
      _message = null;
      _step = 1;
    });
  }

  Future<bool> _onFirstPin(String pin) async {
    setState(() {
      _pin = pin;
      _step = 2;
      _message = null;
    });
    return true;
  }

  Future<bool> _onConfirmPin(String pin) async {
    if (pin != _pin) {
      setState(() {
        _pin = null;
        _step = 1;
        _message = 'PIN не совпал. Придумайте ещё раз.';
      });
      return false;
    }

    final name = _name.text.trim();
    try {
      await ref.read(serviceProvider).createChild(
            nickname: name,
            avatarId: _avatar,
            grade: _grade ?? 1,
            pin: pin,
          );
      if (!mounted) return true;
      showSnack(context, 'Герой $name готов к приключениям! 🎉');
      final navigator = Navigator.of(context);
      final isFirst = widget.isFirst;
      ref.invalidate(childrenProvider);
      if (!isFirst && navigator.canPop()) navigator.pop();
      return true;
    } catch (error) {
      if (mounted) {
        setState(() {
          _step = 1;
          _pin = null;
          _message = friendlyError(error);
        });
      }
      return false;
    }
  }

  @override
  Widget build(BuildContext context) {
    final text = Theme.of(context).textTheme;
    final message = _message;
    final name = _name.text.trim();

    return MagicScaffold(
      title: 'Новый герой',
      showBack: !widget.isFirst,
      actions: widget.isFirst
          ? [
              IconButton(
                tooltip: 'Выйти',
                icon: const Icon(Icons.logout_rounded),
                onPressed: () => ref.read(serviceProvider).signOut(),
              ),
            ]
          : null,
      body: Center(
        child: SingleChildScrollView(
          padding: const EdgeInsets.all(20),
          child: ConstrainedBox(
            constraints: const BoxConstraints(maxWidth: 480),
            child: Column(
              mainAxisSize: MainAxisSize.min,
              children: [
                if (_step == 0) ..._buildDetails(text),
                if (_step == 1) ...[
                  ScivoRaven(avatarId: _avatar, size: 110),
                  const SizedBox(height: 12),
                  Text(
                    'Придумайте PIN для $name',
                    textAlign: TextAlign.center,
                    style: text.titleLarge?.copyWith(fontWeight: FontWeight.w800),
                  ),
                  const SizedBox(height: 6),
                  const Text(
                    '4 цифры. Ребёнок будет вводить их при входе — запишите PIN.',
                    textAlign: TextAlign.center,
                    style: TextStyle(color: Colors.white70),
                  ),
                  const SizedBox(height: 20),
                  PinPad(key: const ValueKey('child-pin-1'), onCompleted: _onFirstPin),
                  TextButton(
                    onPressed: () => setState(() {
                      _step = 0;
                      _message = null;
                    }),
                    child: const Text('Назад'),
                  ),
                ],
                if (_step == 2) ...[
                  ScivoRaven(avatarId: _avatar, size: 110),
                  const SizedBox(height: 12),
                  Text(
                    'Повторите PIN',
                    style: text.titleLarge?.copyWith(fontWeight: FontWeight.w800),
                  ),
                  const SizedBox(height: 20),
                  PinPad(key: const ValueKey('child-pin-2'), onCompleted: _onConfirmPin),
                  TextButton(
                    onPressed: () => setState(() {
                      _step = 1;
                      _pin = null;
                    }),
                    child: const Text('Назад'),
                  ),
                ],
                if (message != null)
                  Padding(
                    padding: const EdgeInsets.only(top: 16),
                    child: Text(
                      message,
                      textAlign: TextAlign.center,
                      style: const TextStyle(
                        color: AppColors.danger,
                        fontWeight: FontWeight.w700,
                      ),
                    ),
                  ),
              ],
            ),
          ),
        ),
      ),
    );
  }

  List<Widget> _buildDetails(TextTheme text) {
    return [
      ScivoSays(
        size: 100,
        avatarId: _avatar,
        text: widget.isFirst
            ? 'Давайте создадим первого героя! Как его зовут?'
            : 'Новый герой? Отлично! Как его зовут?',
      ),
      const SizedBox(height: 20),
      TextField(
        controller: _name,
        maxLength: 20,
        textCapitalization: TextCapitalization.words,
        onChanged: (_) => setState(() {}),
        decoration: magicInput(
          'Имя или никнейм',
          icon: Icons.face_rounded,
          hint: 'Например, Тагир',
        ),
      ),
      const SizedBox(height: 16),
      Align(
        alignment: Alignment.centerLeft,
        child: Text('Класс', style: text.titleMedium?.copyWith(fontWeight: FontWeight.w800)),
      ),
      const SizedBox(height: 8),
      Row(
        children: [
          for (var g = 1; g <= 4; g++)
            Expanded(
              child: Padding(
                padding: const EdgeInsets.all(4),
                child: _GradeButton(
                  grade: g,
                  selected: _grade == g,
                  onTap: () => setState(() => _grade = g),
                ),
              ),
            ),
        ],
      ),
      const SizedBox(height: 16),
      Align(
        alignment: Alignment.centerLeft,
        child: Text(
          'Цвет вороненка',
          style: text.titleMedium?.copyWith(fontWeight: FontWeight.w800),
        ),
      ),
      const SizedBox(height: 8),
      Wrap(
        spacing: 10,
        runSpacing: 10,
        alignment: WrapAlignment.center,
        children: [
          for (final option in RavenArt.avatars)
            _AvatarChoice(
              avatarId: option.id,
              label: option.label,
              selected: _avatar == option.id,
              onTap: () => setState(() => _avatar = option.id),
            ),
        ],
      ),
      const SizedBox(height: 24),
      SizedBox(
        width: double.infinity,
        child: FilledButton(onPressed: _toPin, child: const Text('Дальше')),
      ),
    ];
  }
}

          child: Column(

          ),

      ),
    );
  }
}

}

class _ChildPinScreenState extends ConsumerState<ChildPinScreen> {
  String? _message;

  Future<bool> _onPin(String pin) async {
    try {
      final result =
          await ref.read(serviceProvider).verifyChildPin(widget.kid.id, pin);
      if (!mounted) return false;
      if (result.ok) {
        ref.read(activeChildProvider.notifier).state = widget.kid;
        Navigator.of(context).pop();
        return true;
      }
      setState(() => _message = result.message);
      return false;
    } catch (error) {
      if (mounted) setState(() => _message = friendlyError(error));
      return false;
    }
  }

  @override
  Widget build(BuildContext context) {
    final message = _message;

    return MagicScaffold(
      title: 'Вход',

            constraints: const BoxConstraints(maxWidth: 440),

                ScivoSays(

import 'package:flutter/material.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';

import '../../core/errors.dart';
import '../../core/theme/app_theme.dart';

/// История-знакомство со Scivo. Показывается ребёнку один раз (+20 маны).
class OnboardingScreen extends ConsumerStatefulWidget {
  const OnboardingScreen({
    super.key,
    required this.kid,
    required this.onFinished,
  });

  final ChildProfile kid;
  final void Function(int manaBonus) onFinished;

  @override
  ConsumerState<OnboardingScreen> createState() => _OnboardingScreenState();
}

class _Slide {
  const _Slide(this.emoji, this.text);

  final String emoji;
  final String text;
}

class _OnboardingScreenState extends ConsumerState<OnboardingScreen> {
  int _index = 0;
  bool _sparked = false;
  bool _saving = false;
  String? _error;

  List<_Slide> get _slides => [
        _Slide(
          '📚',
          'Привет, ${widget.kid.nickname}! Я Scivo — вороненок-хранитель '
              'Библиотеки Чисел.',
        ),
        const _Slide(
          '🌀',
          'Беда! Хаос рассыпал магию чисел по пяти островам. Без неё часы '
              'останавливаются, а мосты рушатся.',
        ),
        const _Slide(
          '✨',
          'За каждое задание ты получаешь ману. Мана — это магия: на неё можно '
              'купить подсказку или наряд для меня!',
        ),
        const _Slide(
          '🪶',
          'А перья — особенные! Их дают за задания без подсказок и победу над '
              'боссом. Перья можно обменять на настоящие призы от родителей!',
        ),
        const _Slide(
          '⚡',
          'Нажми на меня, чтобы подарить мне первую искру знаний!',
        ),
      ];

  bool get _isLast => _index == _slides.length - 1;

  Future<void> _finish() async {
    setState(() {
      _saving = true;
      _error = null;
    });
    try {
      final bonus =
          await ref.read(serviceProvider).completeOnboarding(widget.kid.id);
      if (!mounted) return;
      widget.onFinished(bonus);
    } catch (error) {
      if (mounted) {
        setState(() {
          _saving = false;
          _error = friendlyError(error);
        });
      }
    }
  }

  void _next() {
    setState(() => _index = _index + 1);
  }

  void _spark() {
    setState(() => _sparked = true);
  }

  @override
  Widget build(BuildContext context) {
    final slides = _slides;
    final slide = slides[_index];
    final error = _error;

    VoidCallback? mainAction;
    String mainLabel;
    if (_saving) {
      mainAction = null;
      mainLabel = 'Колдую...';
    } else if (!_isLast) {
      mainAction = _next;
      mainLabel = 'Дальше';
    } else if (_sparked) {
      mainAction = _finish;
      mainLabel = 'Начать приключение!';
    } else {
      mainAction = null;
      mainLabel = 'Нажми на Scivo ☝️';
    }

    final bubbleText = _isLast && _sparked
        ? 'Ура! Я чувствую силу! Спасибо, ${widget.kid.nickname}! '
            'Вперёд, к приключениям!'
        : slide.text;

    return MagicScaffold(
      body: Column(
        children: [
          Align(
            alignment: Alignment.topRight,
            child: Padding(
              padding: const EdgeInsets.all(8),
              child: TextButton(
                onPressed: _saving ? null : _finish,
                child: const Text('Пропустить'),
              ),
            ),
          ),
          Expanded(
            child: Center(
              child: SingleChildScrollView(
                padding: const EdgeInsets.symmetric(horizontal: 24),
                child: ConstrainedBox(
                  constraints: const BoxConstraints(maxWidth: 480),
                  child: Column(
                    mainAxisSize: MainAxisSize.min,
                    children: [
                      Text(
                        _isLast && _sparked ? '🌟' : slide.emoji,
                        style: const TextStyle(fontSize: 56),
                      ),
                      const SizedBox(height: 8),
                      GestureDetector(
                        onTap: _isLast && !_sparked ? _spark : null,
                        child: AnimatedScale(
                          scale: _sparked ? 1.15 : 1.0,
                          duration: const Duration(milliseconds: 500),
                          curve: Curves.elasticOut,
                          child: ScivoRaven(size: 170, avatarId: widget.kid.avatarId),
                        ),
                      ),
                      if (_isLast && _sparked)
                        const Padding(
                          padding: EdgeInsets.only(top: 4),
                          child: Text(
                            '+20 ✨',
                            style: TextStyle(
                              fontSize: 28,
                              fontWeight: FontWeight.w900,
                              color: AppColors.gold,
                            ),
                          ),
                        ),
                      const SizedBox(height: 12),
                      AnimatedSwitcher(
                        duration: const Duration(milliseconds: 300),
                        child: SpeechBubble(
                          bubbleText,
                          key: ValueKey('$_index-$_sparked'),
                        ),
                      ),
                    ],
                  ),
                ),
              ),
            ),
          ),
          Padding(
            padding: const EdgeInsets.fromLTRB(24, 8, 24, 20),
            child: ConstrainedBox(
              constraints: const BoxConstraints(maxWidth: 480),
              child: Column(
                mainAxisSize: MainAxisSize.min,
                children: [
                  Row(
                    mainAxisAlignment: MainAxisAlignment.center,
                    children: [
                      for (var i = 0; i < slides.length; i++)
                        AnimatedContainer(
                          duration: const Duration(milliseconds: 250),
                          margin: const EdgeInsets.symmetric(horizontal: 4),
                          width: i == _index ? 22 : 10,
                          height: 10,
                          decoration: BoxDecoration(
                            color: i == _index ? AppColors.gold : Colors.white30,
                            borderRadius: BorderRadius.circular(5),
                          ),
                        ),
                    ],
                  ),
                  const SizedBox(height: 16),
                  if (error != null)
                    Padding(
                      padding: const EdgeInsets.only(bottom: 12),
                      child: Text(
                        error,
                        textAlign: TextAlign.center,
                        style: const TextStyle(
                          color: AppColors.danger,
                          fontWeight: FontWeight.w700,
                        ),
                      ),
                    ),
                  SizedBox(
                    width: double.infinity,
                    child: FilledButton(onPressed: mainAction, child: Text(mainLabel)),
                  ),
                ],
              ),
            ),
          ),
        ],
      ),
    );
  }
}
