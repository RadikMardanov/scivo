# SCIVO — PROJECT PASSPORT (v1.3)

## Decisions (approved by PO)
- Web link (PWA) via GitHub Pages: https://radikmardanov.github.io/scivo/ . Families in Russia, kids 7–10 (grades 1–4).
- Family model: parent = Supabase Auth user (email+password). Children = rows in `children` (nickname, avatar, grade, PIN hashed).
- Parent zone: parental gate (multiplication) + parent PIN (server-verified, lockout).
- No LLM in v1: rule-based "Scivo Brain" (stage 4), LLM-ready JSON. Voice: flutter_tts + speech_to_text (stage 4).
- Graphics: SVG (lib/art), 6 raven avatar colors. Font: Nunito.
- Answers/rewards server-side only (SQL RPC). Client never writes balances.
- Email notifications: Yandex SMTP (stage 5). Data minimization + consent (parents.consent_at).

/// Главный экран ребёнка: онбординг (1 раз) -> ежедневный бонус -> вкладки.
class ChildShell extends ConsumerStatefulWidget {
  const ChildShell({super.key, required this.kid});

  final ChildProfile kid;

  @override
  ConsumerState<ChildShell> createState() => _ChildShellState();
}

class _ChildShellState extends ConsumerState<ChildShell> {
  late bool _onboarded = widget.kid.onboardingDone;
  bool _checkinStarted = false;
  int _tab = 0;

  @override
  void initState() {
    super.initState();
    if (_onboarded) _scheduleCheckin();
  }

  void _scheduleCheckin() {
    if (_checkinStarted) return;
    _checkinStarted = true;
    WidgetsBinding.instance.addPostFrameCallback((_) => _checkin());
  }

  Future<void> _checkin() async {
    final kidId = widget.kid.id;
    try {
      final result = await ref.read(serviceProvider).checkin(kidId);
      if (!mounted) return;
      ref.invalidate(progressProvider(kidId));
      if (result.manaBonus > 0) {
        final days = result.progress.streakDays;
        final freeze = result.usedFreeze ? '\nЗаморозка спасла твою серию! ❄️' : '';
        await showScivoDialog(
          context,
          text: 'С возвращением, ${widget.kid.nickname}!\n'
              'Ежедневный бонус: +${result.manaBonus} ✨\n'
              'Серия: $days ${pluralRu(days, 'день', 'дня', 'дней')} 🔥$freeze',
          button: 'Ура!',
        );
      }
    } catch (error) {
      if (mounted) showSnack(context, friendlyError(error));
    }
  }

  void _onOnboardingFinished(int bonus) {
    setState(() => _onboarded = true);
    ref.read(activeChildProvider.notifier).state =
        widget.kid.copyWith(onboardingDone: true);
    ref.invalidate(childrenProvider);
    ref.invalidate(progressProvider(widget.kid.id));
    _scheduleCheckin();
  }

  @override
  Widget build(BuildContext context) {
    final kid = widget.kid;
    if (!_onboarded) {
      return OnboardingScreen(kid: kid, onFinished: _onOnboardingFinished);
    }

    final progress = ref.watch(progressProvider(kid.id)).valueOrNull;

    return Scaffold(
      body: MagicBackground(
        child: SafeArea(
          child: Column(
            children: [
              _TopBar(kid: kid, progress: progress),
              Expanded(
                child: IndexedStack(
                  index: _tab,
                  children: [
                    WorldMapTab(kid: kid),
                    WishlistTab(kid: kid, progress: progress),
                    ProfileTab(kid: kid, progress: progress),
                  ],
                ),
              ),
            ],
          ),
        ),
      ),
      bottomNavigationBar: NavigationBar(
        backgroundColor: AppColors.nightCard,
        indicatorColor: AppColors.purple,
        selectedIndex: _tab,
        onDestinationSelected: (index) {
          if (index == 1) ref.invalidate(wishlistProvider);
          if (index != 0) ref.invalidate(progressProvider(kid.id));
          setState(() => _tab = index);
        },
        destinations: const [
          NavigationDestination(icon: Icon(Icons.map_rounded), label: 'Игры'),
          NavigationDestination(icon: Icon(Icons.card_giftcard_rounded), label: 'Вишлист'),
          NavigationDestination(icon: Icon(Icons.person_rounded), label: 'Профиль'),
        ],
      ),
    );
  }
}

/// Призы от родителей. Обмен перьев появится на этапе 5.
class WishlistTab extends ConsumerWidget {
  const WishlistTab({super.key, required this.kid, required this.progress});

  final ChildProfile kid;
  final ChildProgress? progress;

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    final itemsAsync = ref.watch(wishlistProvider);
    final have = progress?.availableFeathers ?? 0;

    return itemsAsync.when(
      loading: () => const Center(child: CircularProgressIndicator()),
      error: (error, _) => ErrorView(
        message: friendlyError(error),
        onRetry: () => ref.invalidate(wishlistProvider),
      ),
      data: (items) {
        if (items.isEmpty) {
          return Center(
            child: SingleChildScrollView(
              padding: const EdgeInsets.all(24),
              child: ScivoSays(
                size: 120,
                avatarId: kid.avatarId,
                text: 'Здесь будут призы за перья 🪶. Попроси родителей '
                    'добавить их в родительской зоне!',
              ),
            ),
          );
        }
        return RefreshIndicator(
          onRefresh: () => ref.refresh(wishlistProvider.future),
          child: ListView(
            padding: const EdgeInsets.all(16),
            children: [
              Text(
                'Мои перья: $have 🪶',
                style: const TextStyle(fontSize: 22, fontWeight: FontWeight.w900),
              ),
              const SizedBox(height: 12),
              for (final item in items)
                Padding(
                  padding: const EdgeInsets.only(bottom: 12),
                  child: _PrizeCard(item: item, have: have),
                ),
            ],
          ),
        );
      },
    );
  }
}

class _PrizeCard extends StatelessWidget {
  const _PrizeCard({required this.item, required this.have});

  final WishlistItem item;
  final int have;

  @override
  Widget build(BuildContext context) {
    final cost = item.featherCost;
    final ratio = cost <= 0 ? 1.0 : (have / cost).clamp(0.0, 1.0);
    final enough = have >= cost;
    final left = cost - have;

    return MagicCard(
      padding: const EdgeInsets.all(14),
      child: Row(
        children: [
          Text(item.emoji, style: const TextStyle(fontSize: 40)),
          const SizedBox(width: 12),
          Expanded(
            child: Column(
              crossAxisAlignment: CrossAxisAlignment.start,
              children: [
                Text(
                  item.title,
                  style: const TextStyle(fontSize: 18, fontWeight: FontWeight.w800),
                ),
                Text('$cost 🪶', style: const TextStyle(color: AppColors.gold)),
                const SizedBox(height: 8),
                ClipRRect(
                  borderRadius: BorderRadius.circular(6),
                  child: LinearProgressIndicator(
                    value: ratio,
                    minHeight: 10,
                    backgroundColor: Colors.white12,
                    color: enough ? AppColors.success : AppColors.gold,
                  ),
                ),
                const SizedBox(height: 6),
                Text(
                  enough
                      ? 'Перьев хватает! Обмен появится совсем скоро 🎉'
                      : 'Ещё $left ${pluralRu(left, 'перо', 'пера', 'перьев')}',
                  style: const TextStyle(color: Colors.white70),
                ),
              ],
            ),
          ),
        ],
      ),
    );
  }
}
